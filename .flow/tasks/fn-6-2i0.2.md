### 2. Truncation logic

```tsx
const MAX_COMMENT_LENGTH = 50;

const TruncatedComment = ({ comment, shotId }: Props) => {
  const [expanded, setExpanded] = useState(false);
  const needsTruncation = comment && comment.length > MAX_COMMENT_LENGTH;
  
  return (
    <div>
      <span className={!expanded && needsTruncation ? 'truncate' : ''}>
        {expanded ? comment : comment?.slice(0, MAX_COMMENT_LENGTH)}
        {!expanded && needsTruncation && '...'}
      </span>
      {needsTruncation && (
        <button
          aria-expanded={expanded}
          onClick={() => setExpanded(!expanded)}
          data-testid={`comment-expand-${shotId}`}
        >
          {expanded ? '▼' : '▶'}
        </button>
      )}
    </div>
  );
};
```

### 3. Accessibility

- `aria-expanded` on expand button
- Keyboard: Tab to focus, Enter/Space to toggle
- Focus visible indicator

### 4. data-testid attributes

- `comment-{sectionIdx}-{shotIdx}` on cell
- `comment-expand-{sectionIdx}-{shotIdx}` on expand button
