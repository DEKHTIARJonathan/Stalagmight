# fn-11-6w4.4 Modify SVG generation to use depth colors

## Description
Modify SVG generation to use depth colors with inline styles.

**CRITICAL: ALL station circles (normal + zero datum) get inline `style="fill:..."` in BOTH color modes, for consistency.**

## Station Coloring (both modes use inline fill)
```rust
for station in &stations {
    let fill_color = if station.is_zero_datum {
        "#ffd700".to_string()  // Gold for zero datum, always
    } else {
        match options.color_mode {
            ColorMode::Section => section_color.clone(),
            ColorMode::Depth => get_depth_color(
                station.coordinates.z,
                min_depth,
                max_depth,
                options.depth_preset
            ),
        }
    };
    
    // ALWAYS use style attribute to set fill (both modes)
    svg.push_str(&format!(
        r#"<circle cx="{:.2}" cy="{:.2}" r="{}" style="fill:{}" stroke="#000" stroke-width="1"/>"#,
        station.coordinates.x,
        -station.coordinates.y,
        radius,
        fill_color
    ));
}
```

## Update `generate_svg_string()`

Modify `src-tauri/src/export/image.rs`:

```rust
use super::depth_colors::{get_depth_color, DepthPreset};

fn generate_svg_string(project: &Project, options: &ExportImageOptions) -> Result<String> {
    let stations = project.compute_station_coordinates();
    
    // Calculate depth range for gradient
    let (min_depth, max_depth) = if options.color_mode == ColorMode::Depth {
        let depths: Vec<f64> = stations.iter().map(|s| s.coordinates.z).collect();
        let min = depths.iter().cloned().fold(f64::INFINITY, f64::min);
        let max = depths.iter().cloned().fold(f64::NEG_INFINITY, f64::max);
        (min, max)
    } else {
        (0.0, 0.0) // Not used for section mode
    };
    
    // ... existing bounds calculation ...
    
    // Draw shots with depth or section color
    for section in &project.sections {
        for shot in &section.shots {
            // Get from/to station depths
            let from_station = find_station(&stations, section.id, shot.from_index);
            let to_station = find_station(&stations, section.id, shot.to_index);
            
            let color = match options.color_mode {
                ColorMode::Section => section.color.clone(),
                ColorMode::Depth => {
                    // Use average depth of shot endpoints
                    let avg_depth = (from_station.coordinates.z + to_station.coordinates.z) / 2.0;
                    get_depth_color(avg_depth, min_depth, max_depth, options.depth_preset)
                }
            };
            
            svg.push_str(&format!(
                r#"<line class="shot" x1="{:.2}" y1="{:.2}" x2="{:.2}" y2="{:.2}" stroke="{}"/>"#,
                from_station.coordinates.x, -from_station.coordinates.y,
                to_station.coordinates.x, -to_station.coordinates.y,
                color
            ));
        }
    }
    
    // Draw stations with depth or section color
    for station in &stations {
        let section = project.sections.iter().find(|s| s.id == station.station_ref.section_id);
        
        let fill_color = if station.is_zero_datum {
            "#ffd700".to_string() // Gold for zero datum, always
        } else {
            match options.color_mode {
                ColorMode::Section => section.map(|s| s.color.clone()).unwrap_or("#ffffff".to_string()),
                ColorMode::Depth => get_depth_color(
                    station.coordinates.z,
                    min_depth,
                    max_depth,
                    options.depth_preset
                ),
            }
        };
        
        // ... draw station circle with fill_color ...
    }
    
    // ... rest of SVG generation ...
}
```
