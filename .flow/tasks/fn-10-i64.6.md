function parseLoadError(e: unknown): LoadError {
  if (typeof e === 'string') {
    try {
      return JSON.parse(e) as LoadError;
    } catch {
      return { code: 'UNKNOWN', message: e };
    }
  }
  if (typeof e === 'object' && e !== null) {
    const obj = e as any;
    return {
      code: obj.code || 'UNKNOWN',

      details: obj.details,
    };

}

loadProject: async (path: string) => {
  // IMPORTANT: Explicitly cancel any existing load first
  // This ensures clean state before starting new load
  if (get().loading.isLoading) {
    await get().cancelLoading();
  }
  
  const initialName = path.split('/').pop()?.replace(/\.(cvproj|cvchef)$/, '') ?? 'Project';
  const loadingId = get().startLoading(initialName, path);
  
  try {
    const result = await invoke<LoadProjectFullResult>('load_project_full', { path });
    
    if (get().loading.loadingId !== loadingId) {
      return { success: false };
    }
    
    // Update loading title with actual project name from result
    if (result.project.name && result.project.name !== initialName) {
      get().setLoadingProgress({ projectName: result.project.name });
    }
    
    set({
      project: result.project,
      stations: result.stations,
      stats: result.stats,
      warnings: result.warnings,
      isDirty: false,
    });
    
    // Apply display settings (merge, don't overwrite)
    useUIStore.getState().applyDisplaySettings(result.displaySettings);
    useUIStore.getState().triggerFitToBounds();
    
    // Set view to single project view
    useUIStore.getState().setView('single');
    
    await get().loadRecentProjects();
    
    get().resetLoading();
    return { success: true };
  } catch (e) {
    const error = parseLoadError(e);
    
    if (error.code === 'ALREADY_OPEN') {
      useUIStore.getState().setView('single');
      get().resetLoading();
      return { success: true };
    }
    
    if (get().loading.loadingId === loadingId) {
      get().setLoadingError(error.message, error.details);
    }
    return { success: false };
  }
},

```typescript
import { useEffect, useRef } from 'react';
import { listen, type UnlistenFn } from '@tauri-apps/api/event';
import { useProjectStore } from '../store/projectStore';

interface LoadingProgressPayload {
  loadingId: number;
  phase: string;
  phaseIndex: number;
  phaseCount: number;
  current: number | null;
  total: number | null;
  overallPercent: number;
  isIndeterminate: boolean;
}

const TIMEOUT_MS = 30000;

export function useLoadingProgress() {
  const setLoadingProgress = useProjectStore((s) => s.setLoadingProgress);
  const loading = useProjectStore((s) => s.loading);
  const timeoutRef = useRef<number | null>(null);

  useEffect(() => {
    if (!loading.isLoading || loading.error) return;

    let unlisten: UnlistenFn | null = null;

    const checkTimeout = () => {
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
      timeoutRef.current = window.setTimeout(() => {
        const current = useProjectStore.getState().loading;
        if (current.isLoading && !current.error) {
          setLoadingProgress({ phase: 'loading_taking_long' });
        }
      }, TIMEOUT_MS);
    };

    const setupListener = async () => {
      unlisten = await listen<LoadingProgressPayload>('loading-progress', (event) => {
        const currentLoadingId = useProjectStore.getState().loading.loadingId;
        
        // CRITICAL: Ignore stale events
        if (event.payload.loadingId !== currentLoadingId) {
          if (import.meta.env.DEV) {
            console.debug('[Loading] Ignoring stale event', event.payload.loadingId, '!=', currentLoadingId);
          }
          return;
        }

        checkTimeout();

        const { phase, phaseIndex, phaseCount, current, total, overallPercent, isIndeterminate } = event.payload;
        setLoadingProgress({
          phase,
          phaseIndex,
          phaseCount,
          phaseProgress: isIndeterminate ? undefined : (current && total ? (current / total) * 100 : 0),
          phaseCurrent: current ?? undefined,
          phaseTotal: total ?? undefined,
          overallProgress: overallPercent,
        });
      });
    };

    checkTimeout();
    setupListener();

    return () => {
      unlisten?.();
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
    };
  }, [loading.isLoading, loading.error, setLoadingProgress]);
}
```
