      return { success: true };

    }
    return { success: false };
  }
},

## New File: src/hooks/useLoadingProgress.ts

```typescript
import { useEffect, useRef } from 'react';
import { listen, type UnlistenFn } from '@tauri-apps/api/event';
import { useProjectStore } from '../store/projectStore';

interface LoadingProgressPayload {
  loadingId: number;
  phase: string;
  phaseIndex: number;
  phaseCount: number;
  current: number | null;
  total: number | null;
  overallPercent: number;
  isIndeterminate: boolean;
}

const TIMEOUT_MS = 30000;

export function useLoadingProgress() {
  const setLoadingProgress = useProjectStore((s) => s.setLoadingProgress);
  const loading = useProjectStore((s) => s.loading);
  const timeoutRef = useRef<number | null>(null);

  useEffect(() => {
    if (!loading.isLoading || loading.error) return;

    let unlisten: UnlistenFn | null = null;

    const checkTimeout = () => {
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
      timeoutRef.current = window.setTimeout(() => {
        const current = useProjectStore.getState().loading;
        if (current.isLoading && !current.error) {
          setLoadingProgress({ phase: 'loading_taking_long' });
        }
      }, TIMEOUT_MS);
    };

    const setupListener = async () => {
      unlisten = await listen<LoadingProgressPayload>('loading-progress', (event) => {
        const currentLoadingId = useProjectStore.getState().loading.loadingId;
        
        // CRITICAL: Ignore stale events
        if (event.payload.loadingId !== currentLoadingId) {
          if (import.meta.env.DEV) {
            console.debug('[Loading] Ignoring stale event', event.payload.loadingId, '!=', currentLoadingId);
          }
          return;
        }

        checkTimeout();

        const { phase, phaseIndex, phaseCount, current, total, overallPercent, isIndeterminate } = event.payload;
        setLoadingProgress({
          phase,
          phaseIndex,
          phaseCount,
          phaseProgress: isIndeterminate ? undefined : (current && total ? (current / total) * 100 : 0),
          phaseCurrent: current ?? undefined,
          phaseTotal: total ?? undefined,
          overallProgress: overallPercent,
        });
      });
    };

    checkTimeout();
    setupListener();

    return () => {
      unlisten?.();
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
    };
  }, [loading.isLoading, loading.error, setLoadingProgress]);
}
```

import { useEffect, useRef } from 'react';
import { listen, type UnlistenFn } from '@tauri-apps/api/event';
import { useProjectStore } from '../store/projectStore';

interface LoadingProgressPayload {
  loadingId: number;
  phase: string;
  phaseIndex: number;
  phaseCount: number;
  current: number | null;
  total: number | null;
  overallPercent: number;
  isIndeterminate: boolean;
}

const TIMEOUT_MS = 30000;

export function useLoadingProgress() {
  const setLoadingProgress = useProjectStore((s) => s.setLoadingProgress);
  const loading = useProjectStore((s) => s.loading);
  const timeoutRef = useRef<number | null>(null);

  useEffect(() => {
    if (!loading.isLoading || loading.error) return;

    let unlisten: UnlistenFn | null = null;

    const checkTimeout = () => {
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
      
      // Only timeout if NOT in indeterminate phase (heartbeat handles those)
      timeoutRef.current = window.setTimeout(() => {
        const current = useProjectStore.getState().loading;
        const timeSinceEvent = Date.now() - (current.lastEventTime || 0);
        
        if (timeSinceEvent > TIMEOUT_MS && current.isLoading && !current.error) {
          setLoadingProgress({
            phase: 'loading_taking_long',
          });
        }
      }, TIMEOUT_MS);
    };

    const setupListener = async () => {
      unlisten = await listen<LoadingProgressPayload>('loading-progress', (event) => {
        const currentLoadingId = useProjectStore.getState().loading.loadingId;
        
        // CRITICAL: Ignore stale events from cancelled/previous loads
        if (event.payload.loadingId !== currentLoadingId) {
          if (import.meta.env.DEV) {
            console.debug('[Loading] Ignoring stale event', event.payload.loadingId, '!=', currentLoadingId);
          }
          return;
        }

        checkTimeout(); // Reset timeout

        const { phase, phaseIndex, phaseCount, current, total, overallPercent, isIndeterminate } = event.payload;

        setLoadingProgress({
          phase,
          phaseIndex,
          phaseCount,
          phaseProgress: isIndeterminate ? undefined : (current && total ? (current / total) * 100 : 0),
          phaseCurrent: current ?? undefined,
          phaseTotal: total ?? undefined,
          overallProgress: overallPercent,
        });

        if (import.meta.env.DEV) {
          console.debug('[Loading]', phase, phaseIndex+1, '/', phaseCount, current, '/', total, `(${overallPercent.toFixed(1)}%)`);
        }
      });
    };

    checkTimeout();
    setupListener();

    return () => {
      unlisten?.();
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
    };
  }, [loading.isLoading, loading.error, setLoadingProgress]);
}

```typescript
import { useEffect, useRef } from 'react';
import { listen, type UnlistenFn } from '@tauri-apps/api/event';
import { useProjectStore } from '../store/projectStore';

interface LoadingProgressPayload {
  phase: string;
  phaseLabel: string;
  current: number | null;
  total: number | null;
  overallPercent: number;
  isIndeterminate: boolean;
}

const TIMEOUT_MS = 30000; // 30 second timeout

export function useLoadingProgress() {
  const setLoadingProgress = useProjectStore((s) => s.setLoadingProgress);
  const isLoading = useProjectStore((s) => s.loading.isLoading);
  const timeoutRef = useRef<number | null>(null);

  useEffect(() => {
    if (!isLoading) return;

    let unlisten: UnlistenFn | null = null;

    const resetTimeout = () => {
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
      timeoutRef.current = window.setTimeout(() => {
        setLoadingProgress({
          phase: 'Loading is taking longer than expected...',
        });
      }, TIMEOUT_MS);
    };

    const setupListener = async () => {
      unlisten = await listen<LoadingProgressPayload>('loading-progress', (event) => {
        const { phaseLabel, current, total, overallPercent, isIndeterminate } = event.payload;

        resetTimeout(); // Reset timeout on each progress event

        setLoadingProgress({
          phase: phaseLabel,
          phaseProgress: isIndeterminate ? undefined : (current && total ? (current / total) * 100 : 0),
          phaseCurrent: current ?? undefined,
          phaseTotal: total ?? undefined,
          overallProgress: overallPercent,
        });

        // Debug logging
        if (import.meta.env.DEV) {
          console.debug('[Loading]', phaseLabel, current, '/', total, `(${overallPercent.toFixed(1)}%)`);
        }
      });
    };

    resetTimeout(); // Start initial timeout
    setupListener();

    return () => {
      unlisten?.();
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
    };
  }, [isLoading, setLoadingProgress]);
}
```
