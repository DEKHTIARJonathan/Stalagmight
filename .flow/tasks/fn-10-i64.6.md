```typescript
interface LoadingState {
  isLoading: boolean;
  projectName?: string;
  phase?: string;
  phaseProgress?: number;
  phaseCurrent?: number;
  phaseTotal?: number;
  overallProgress: number;
  error?: { message: string; details?: string };
}

interface ProjectStore {
  // ... existing fields
  loading: LoadingState;
  setLoadingProgress: (progress: Partial<LoadingState>) => void;
  resetLoading: () => void;
  setLoadingError: (message: string, details?: string) => void;
}

const initialLoadingState: LoadingState = {
  isLoading: false,
  overallProgress: 0,
};

// In create():
loading: { ...initialLoadingState },

setLoadingProgress: (progress) => set((state) => ({
  loading: { ...state.loading, ...progress }
})),

resetLoading: () => set({ loading: { ...initialLoadingState } }),

setLoadingError: (message, details) => set((state) => ({
  loading: { ...state.loading, error: { message, details } }
})),
```

## New File: src/hooks/useLoadingProgress.ts

Hook to listen for Tauri loading-progress events:

```typescript
import { useEffect } from 'react';
import { listen, UnlistenFn } from '@tauri-apps/api/event';
import { useProjectStore } from '../store/projectStore';

interface LoadingProgressPayload {
  phase: string;
  phase_label: string;
  current: number | null;
  total: number | null;
  overall_percent: number;
  is_indeterminate: boolean;
}

export function useLoadingProgress() {
  const setLoadingProgress = useProjectStore((s) => s.setLoadingProgress);

  useEffect(() => {
    let unlisten: UnlistenFn | null = null;

    const setupListener = async () => {
      unlisten = await listen<LoadingProgressPayload>('loading-progress', (event) => {
        const { phase_label, current, total, overall_percent, is_indeterminate } = event.payload;

        setLoadingProgress({
          phase: phase_label,
          phaseProgress: is_indeterminate ? undefined : (current && total ? (current / total) * 100 : 0),
          phaseCurrent: current ?? undefined,
          phaseTotal: total ?? undefined,
          overallProgress: overall_percent,
        });

        // Debug logging
        if (import.meta.env.DEV) {
          console.debug('[Loading]', phase_label, current, '/', total, `(${overall_percent.toFixed(1)}%)`);
        }
      });
    };

    setupListener();

    return () => {
      unlisten?.();
    };
  }, [setLoadingProgress]);
}
```
