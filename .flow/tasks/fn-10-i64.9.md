// Add to mock setup
const eventListeners = new Map<string, Set<(event: any) => void>>();

// Mock listen function
async function mockListen(event: string, handler: (event: any) => void) {
  if (!eventListeners.has(event)) {
    eventListeners.set(event, new Set());
  }
  eventListeners.get(event)!.add(handler);
  
  // Return unlisten function
  return () => {
    eventListeners.get(event)?.delete(handler);
  };
}

// Helper to emit events from tests
function emitMockEvent(event: string, payload: any) {
  const listeners = eventListeners.get(event);
  if (listeners) {
    listeners.forEach(handler => handler({ payload }));
  }
}
