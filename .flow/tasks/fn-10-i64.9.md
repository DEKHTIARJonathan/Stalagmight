## Unit Tests: src/components/LoadingProgress/__tests__/LoadingProgress.test.tsx

```typescript
import { render, screen, fireEvent } from '@testing-library/react';
import { LoadingProgress } from '../LoadingProgress';

describe('LoadingProgress', () => {
  const defaultProps = {
    title: 'Loading Test Project...',
    phase: 'Loading survey data...',
    overallProgress: 45,
  };

  it('renders title and phase', () => {
    render(<LoadingProgress {...defaultProps} />);
    expect(screen.getByText('Loading Test Project...')).toBeInTheDocument();
    expect(screen.getByText('Loading survey data...')).toBeInTheDocument();
  });

  it('shows overall progress percentage', () => {
    render(<LoadingProgress {...defaultProps} />);
    expect(screen.getByText('45%')).toBeInTheDocument();
  });

  it('shows item counts when provided', () => {
    render(<LoadingProgress {...defaultProps} phaseCurrent={23} phaseTotal={50} />);
    expect(screen.getByText('23/50')).toBeInTheDocument();
  });

  it('shows indeterminate state when phaseProgress undefined', () => {
    render(<LoadingProgress {...defaultProps} />);
    const indeterminateBar = document.querySelector('.animate-pulse-shimmer');
    expect(indeterminateBar).toBeInTheDocument();
  });

  it('shows determinate progress bar when phaseProgress provided', () => {
    render(<LoadingProgress {...defaultProps} phaseProgress={60} />);
    const indeterminateBar = document.querySelector('.animate-pulse-shimmer');
    expect(indeterminateBar).not.toBeInTheDocument();
  });

  it('calls onCancel when cancel clicked', () => {
    const onCancel = jest.fn();
    render(<LoadingProgress {...defaultProps} onCancel={onCancel} />);
    fireEvent.click(screen.getByText('Cancel'));
    expect(onCancel).toHaveBeenCalled();
  });

  it('renders error state correctly', () => {
    render(
      <LoadingProgress 
        {...defaultProps} 
        error={{ message: 'File not found', details: 'ENOENT' }}
        onCancel={() => {}}
      />
    );
    expect(screen.getByText('Failed to Load Project')).toBeInTheDocument();
    expect(screen.getByText('File not found')).toBeInTheDocument();
    expect(screen.getByText('Dismiss')).toBeInTheDocument();
  });

  it('has correct ARIA attributes', () => {
    render(<LoadingProgress {...defaultProps} />);
    expect(screen.getByRole('dialog')).toBeInTheDocument();
    expect(screen.getByRole('status')).toBeInTheDocument();
  });
});
```

## E2E Tests: e2e-playwright/loading-progress.spec.ts

```typescript
import { test, expect, Page } from '@playwright/test';
import { setupTauriMocks } from './mocks/tauri-ipc';

test.describe('Loading Progress', () => {
  test.beforeEach(async ({ page }) => {
    await setupTauriMocks(page);
    await page.goto('/');
    await page.waitForLoadState('networkidle');
  });

  test('shows progress UI when opening project', async ({ page }) => {
    // Click on a project to open it
    const projectCard = page.locator('[data-testid="project-card"]').first();
    await projectCard.dblclick();

    // Progress UI should appear
    await expect(page.locator('[data-testid="loading-progress"]')).toBeVisible();
  });

  test('shows project name in loading title', async ({ page }) => {
    await page.locator('[data-testid="project-card"]').first().dblclick();
    await expect(page.getByText(/Loading .+\.\.\./)).toBeVisible();
  });

  test('shows phase updates', async ({ page }) => {
    await page.locator('[data-testid="project-card"]').first().dblclick();
    // Phase text should update
    await expect(page.getByText(/Loading|Computing|Validating/)).toBeVisible();
  });

  test('cancel button returns to GlobalView', async ({ page }) => {
    await page.locator('[data-testid="project-card"]').first().dblclick();
    await page.getByText('Cancel').click();
    // Should be back at GlobalView
    await expect(page.locator('[data-testid="global-view"]')).toBeVisible();
  });
});
```

## E2E Testing Hooks

Add to window.__E2E_TEST__:

```typescript
// In projectStore or App.tsx
if (import.meta.env.DEV) {
  (window as any).__E2E_TEST__ = {
    ...existing,
    simulateLoadingProgress: (phase: string, current: number, total: number, overall: number) => {
      useProjectStore.getState().setLoadingProgress({
        phase,
        phaseCurrent: current,
        phaseTotal: total,
        phaseProgress: (current / total) * 100,
        overallProgress: overall,
      });
    },
    simulateLoadingError: (message: string, details?: string) => {
      useProjectStore.getState().setLoadingError(message, details);
    },
  };
}
```
