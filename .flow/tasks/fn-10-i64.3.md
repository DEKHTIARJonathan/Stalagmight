#[tauri::command]
pub async fn create_project_full(
    state: State<'_, AppState>,
    name: String,
    location: Option<PathBuf>
) -> Result<CreateProjectFullResult, String> {
    let state_clone = state.inner().clone();

    tauri::async_runtime::spawn_blocking(move || {
        state_clone.do_create_project_full(name, location)

## Implementation in state.rs

```rust
impl AppState {
    pub fn do_create_project_full(
        &self,
        name: String,
        location: Option<PathBuf>
    ) -> Result<CreateProjectFullResult, String> {
        let handle = self.get_app_handle().ok_or("AppHandle not initialized")?;
        let cancel_token = self.start_new_load();
        let mut progress = ProgressEmitter::new_for_create(handle, cancel_token.clone());

        // Phase 0: Creating database (40%)
        progress.start_phase(LoadingPhase::CreatingDatabase);
        let path = self.resolve_project_path(&name, location)?;
        let db = Database::create(&path).map_err(|e| e.to_string())?;
        if progress.is_cancelled() { return Err("Cancelled".to_string()); }
        progress.complete_phase(LoadingPhase::CreatingDatabase);

        // Phase 1: Initializing project (40%)
        progress.start_phase(LoadingPhase::InitializingProject);
        let project = Project::new(name.clone());
        db.save_project(&project).map_err(|e| e.to_string())?;
        if progress.is_cancelled() { return Err("Cancelled".to_string()); }
        progress.complete_phase(LoadingPhase::InitializingProject);

        // Phase 2: Saving settings (20%)
        progress.start_phase(LoadingPhase::SavingSettings);
        let display_settings = DisplaySettings::default();
        db.save_display_settings(&display_settings).map_err(|e| e.to_string())?;
        progress.complete_phase(LoadingPhase::SavingSettings);

        // Store project, update recent
        *self.project.lock().unwrap() = Some(project.clone());
        self.add_to_recent(&path)?;

        Ok(CreateProjectFullResult {
            loading_id: progress.loading_id(),
            project,
            display_settings,
        })
    }
}
```

| InitializingProject | 40% | Create project record |
| SavingSettings | 20% | Save default display settings |

All phases are indeterminate (no item counts needed - creation is fast).
