```rust
fn create_project(&self, app_handle: AppHandle, name: String, path: PathBuf) -> CommandResult {
    let mut progress = ProgressEmitter::new(app_handle);

    // Phase 1: Creating database (40% weight)
    progress.start_phase(LoadingPhase::CreatingDatabase);
    let db = match Database::create(&path) {
        Ok(db) => db,
        Err(e) => return CommandResult::failure(format!("Failed to create database: {}", e)),
    };
    progress.complete_phase(LoadingPhase::CreatingDatabase);

    // Phase 2: Initializing project (40% weight)
    progress.start_phase(LoadingPhase::InitializingProject);
    let project = Project::new(name);
    // ... initialization logic
    progress.complete_phase(LoadingPhase::InitializingProject);

    // Phase 3: Saving settings (20% weight)
    progress.start_phase(LoadingPhase::SavingSettings);
    // ... save to database
    progress.complete_phase(LoadingPhase::SavingSettings);

    CommandResult::success()
}
```
