use serde::{Deserialize, Serialize};
use tauri::{AppHandle, Emitter};

#[derive(Clone, Serialize, Deserialize, Debug)]
#[serde(rename_all = "snake_case")]
pub enum LoadingPhase {
    // Project opening phases
    OpeningDatabase,
    LoadingProjectData,
    LoadingSections,
    ComputingStations,
    ComputingStatistics,
    ValidatingData,
    LoadingSettings,
    Finalizing,
    // Project creation phases
    CreatingDatabase,
    InitializingProject,
    SavingSettings,
}

impl LoadingPhase {
    pub fn label(&self) -> &'static str {
        match self {
            Self::OpeningDatabase => "Opening database...",
            Self::LoadingProjectData => "Loading survey data...",
            Self::LoadingSections => "Loading sections...",
            Self::ComputingStations => "Calculating station positions...",
            Self::ComputingStatistics => "Computing statistics...",
            Self::ValidatingData => "Validating data...",
            Self::LoadingSettings => "Loading display settings...",
            Self::Finalizing => "Finalizing...",
            Self::CreatingDatabase => "Creating database...",
            Self::InitializingProject => "Initializing project...",
            Self::SavingSettings => "Saving settings...",
        }
    }

    /// Weight for overall progress calculation (out of 100)
    pub fn weight(&self) -> f32 {
        match self {
            Self::OpeningDatabase => 5.0,
            Self::LoadingProjectData => 25.0,
            Self::LoadingSections => 15.0,
            Self::ComputingStations => 20.0,
            Self::ComputingStatistics => 15.0,
            Self::ValidatingData => 10.0,
            Self::LoadingSettings => 5.0,
            Self::Finalizing => 5.0,
            Self::CreatingDatabase => 40.0,
            Self::InitializingProject => 40.0,
            Self::SavingSettings => 20.0,
        }
    }

    pub fn is_measurable(&self) -> bool {
        matches!(
            self,
            Self::LoadingProjectData
                | Self::LoadingSections
                | Self::ComputingStations
                | Self::ComputingStatistics
        )
    }
}

#[derive(Clone, Serialize)]
pub struct LoadingProgress {
    pub phase: LoadingPhase,
    pub phase_label: String,
    pub current: Option<u32>,
    pub total: Option<u32>,
    pub overall_percent: f32,
    pub is_indeterminate: bool,
}

pub struct ProgressEmitter {
    app_handle: AppHandle,
    completed_weight: f32,
    current_phase_weight: f32,
}

impl ProgressEmitter {
    pub fn new(app_handle: AppHandle) -> Self {
        Self {
            app_handle,
            completed_weight: 0.0,
            current_phase_weight: 0.0,
        }
    }

    pub fn start_phase(&mut self, phase: LoadingPhase) {
        self.current_phase_weight = phase.weight();
        self.emit(phase, None, None);
    }

    pub fn update_phase(&self, phase: LoadingPhase, current: u32, total: u32) {
        self.emit(phase, Some(current), Some(total));
    }

    pub fn complete_phase(&mut self, phase: LoadingPhase) {
        self.completed_weight += phase.weight();
        // Emit 100% for this phase
        self.emit(phase, Some(1), Some(1));
    }

    fn emit(&self, phase: LoadingPhase, current: Option<u32>, total: Option<u32>) {
        let phase_progress = match (current, total) {
            (Some(c), Some(t)) if t > 0 => (c as f32 / t as f32) * 100.0,
            _ => 0.0,
        };

        let phase_contribution = (phase_progress / 100.0) * self.current_phase_weight;
        let overall = self.completed_weight + phase_contribution;

        let progress = LoadingProgress {
            phase_label: phase.label().to_string(),
            is_indeterminate: !phase.is_measurable() && current.is_none(),
            phase,
            current,
            total,
            overall_percent: overall.min(100.0),
        };

        let _ = self.app_handle.emit("loading-progress", progress);
    }
}
