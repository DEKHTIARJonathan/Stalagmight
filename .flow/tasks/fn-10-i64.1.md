const THROTTLE_ITEMS: u32 = 10;
const THROTTLE_MS: u64 = 50;
const HEARTBEAT_MS: u64 = 2000;

pub struct ProgressEmitter {
    app_handle: AppHandle,
    loading_id: u64,
    phase_count: u8,
    completed_weight: f32,
    current_phase_weight: f32,
    current_phase_index: u8,
    last_emit: Instant,
    last_emitted_item: u32,
}

impl ProgressEmitter {
    pub fn new_for_open(app_handle: AppHandle) -> Self {
        Self {
            app_handle,
            loading_id: next_loading_id(),
            phase_count: 8,
            completed_weight: 0.0,
            current_phase_weight: 0.0,
            current_phase_index: 0,
            last_emit: Instant::now(),
            last_emitted_item: 0,
        }
    }

    pub fn new_for_create(app_handle: AppHandle) -> Self {
        Self {
            app_handle,
            loading_id: next_loading_id(),
            phase_count: 3,
            completed_weight: 0.0,
            current_phase_weight: 0.0,
            current_phase_index: 0,
            last_emit: Instant::now(),
            last_emitted_item: 0,
        }
    }

    pub fn loading_id(&self) -> u64 {
        self.loading_id
    }

    pub fn start_phase(&mut self, phase: LoadingPhase) {
        self.current_phase_weight = phase.weight();
        self.current_phase_index = phase.index_for_open();
        self.last_emit = Instant::now();
        self.last_emitted_item = 0;
        self.emit(phase, None, None);
    }

    pub fn update_phase(&mut self, phase: LoadingPhase, current: u32, total: u32) {
        let is_boundary = current == 0 || current == total;
        let items_since = current.saturating_sub(self.last_emitted_item);
        let time_since = self.last_emit.elapsed().as_millis() as u64;

        if is_boundary || items_since >= THROTTLE_ITEMS || time_since >= THROTTLE_MS {
            self.emit(phase, Some(current), Some(total));
            self.last_emit = Instant::now();
            self.last_emitted_item = current;
        }
    }

    /// Emit heartbeat for indeterminate phases (call periodically)
    pub fn heartbeat(&mut self, phase: LoadingPhase) {
        if self.last_emit.elapsed() >= Duration::from_millis(HEARTBEAT_MS) {
            self.emit(phase, None, None);
            self.last_emit = Instant::now();
        }
    }

    pub fn complete_phase(&mut self, phase: LoadingPhase) {
        self.completed_weight += phase.weight();
        self.emit(phase, Some(1), Some(1));
    }

    fn emit(&self, phase: LoadingPhase, current: Option<u32>, total: Option<u32>) {
        let phase_progress = match (current, total) {
            (Some(c), Some(t)) if t > 0 => (c as f32 / t as f32) * 100.0,
            _ => 0.0,
        };

        let phase_contribution = (phase_progress / 100.0) * self.current_phase_weight;
        let overall = self.completed_weight + phase_contribution;

        let progress = LoadingProgress {
            loading_id: self.loading_id,
            phase_index: self.current_phase_index,
            phase_count: self.phase_count,
            is_indeterminate: !phase.is_measurable() && current.is_none(),
            phase,
            current,
            total,
            overall_percent: overall.clamp(0.0, 100.0),
        };

        if let Err(e) = self.app_handle.emit("loading-progress", &progress) {
            tracing::warn!("Failed to emit loading progress: {}", e);
        }
    }
}
