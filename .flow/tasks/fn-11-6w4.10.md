import { describe, it, expect } from 'vitest';
import { computeStationColor, computeShotColor, calculateDepthRange } from './export-colors';

describe('calculateDepthRange', () => {
  it('returns default range for empty stations', () => {
    expect(calculateDepthRange([])).toEqual({ min: 0, max: 50 });
  });

  it('calculates range from station z coordinates', () => {
    const stations = [
      { coordinates: { z: 5 } },
      { coordinates: { z: 15 } },
      { coordinates: { z: 10 } },
    ];
    expect(calculateDepthRange(stations)).toEqual({ min: 5, max: 15 });
  });
});

describe('computeStationColor', () => {
  const presetColors = { shallow: '#00bfff', deep: '#ff4444' };
  const depthRange = { min: 0, max: 100 };

  it('returns gold for zero datum regardless of mode', () => {
    const station = { coordinates: { z: 50 }, is_zero_datum: true };
    
    expect(computeStationColor(station, '#ffffff', 'section', depthRange, null)).toBe('#ffd700');
    expect(computeStationColor(station, '#ffffff', 'depth', depthRange, presetColors)).toBe('#ffd700');
  });

  it('returns section color in section mode', () => {
    const station = { coordinates: { z: 50 }, is_zero_datum: false };
    expect(computeStationColor(station, '#ff0000', 'section', depthRange, presetColors)).toBe('#ff0000');
  });

  it('returns depth color in depth mode', () => {
    const station = { coordinates: { z: 0 }, is_zero_datum: false };
    const result = computeStationColor(station, '#ffffff', 'depth', depthRange, presetColors);
    expect(result).toBe('#00bfff'); // shallow color at min depth
  });

  it('returns section color when presetColors is null', () => {
    const station = { coordinates: { z: 50 }, is_zero_datum: false };
    expect(computeStationColor(station, '#ff0000', 'depth', depthRange, null)).toBe('#ff0000');
  });
});

describe('computeShotColor', () => {
  const presetColors = { shallow: '#00bfff', deep: '#ff4444' };
  const depthRange = { min: 0, max: 100 };

  it('returns section color in section mode', () => {
    expect(computeShotColor(0, 50, '#ff0000', 'section', depthRange, presetColors)).toBe('#ff0000');
  });

  it('uses average depth for depth mode', () => {
    // from=0, to=100 -> avg=50, middle of range
    const result = computeShotColor(0, 100, '#ffffff', 'depth', depthRange, presetColors);
    expect(result).not.toBe('#00bfff'); // not shallow
    expect(result).not.toBe('#ff4444'); // not deep
  });
});

import { describe, it, expect, vi } from 'vitest';
import { render, screen, fireEvent } from '@testing-library/react';
import ExportDialog from './ExportDialog';

// Mock stores and Tauri invoke
vi.mock('../../store/uiStore', () => ({
  useUIStore: () => ({
    closeDialog: vi.fn(),
  }),
}));

vi.mock('../../store/projectStore', () => ({
  useProjectStore: () => ({
    exportCsv: vi.fn(),
    exportJson: vi.fn(),
    exportImage: vi.fn(),
  }),
}));

vi.mock('@tauri-apps/api/core', () => ({
  invoke: vi.fn().mockResolvedValue({ shallow: '#00bfff', deep: '#ff4444' }),
}));

describe('ExportDialog', () => {
  it('shows color mode options for image export', () => {
    render(<ExportDialog />);
    
    // Switch to image export
    fireEvent.click(screen.getByTestId('export-type-image'));
    
    expect(screen.getByTestId('color-mode-section')).toBeInTheDocument();
    expect(screen.getByTestId('color-mode-depth')).toBeInTheDocument();
  });

  it('shows depth preset dropdown when depth mode selected', () => {
    render(<ExportDialog />);
    
    fireEvent.click(screen.getByTestId('export-type-image'));
    fireEvent.click(screen.getByTestId('color-mode-depth'));
    
    expect(screen.getByTestId('depth-preset-select')).toBeInTheDocument();
  });

  it('shows include legend checkbox when depth mode selected', () => {
    render(<ExportDialog />);
    
    fireEvent.click(screen.getByTestId('export-type-image'));
    fireEvent.click(screen.getByTestId('color-mode-depth'));
    
    expect(screen.getByTestId('option-include-legend')).toBeInTheDocument();
  });

  it('includes 600 DPI option in resolution selector', () => {
    render(<ExportDialog />);
    
    fireEvent.click(screen.getByTestId('export-type-image'));
    
    const resolutionSelect = screen.getByTestId('resolution-select');
    expect(resolutionSelect).toContainHTML('600');
  });

  it('hides depth options when section mode selected', () => {
    render(<ExportDialog />);
    
    fireEvent.click(screen.getByTestId('export-type-image'));
    fireEvent.click(screen.getByTestId('color-mode-depth'));
    fireEvent.click(screen.getByTestId('color-mode-section'));
    
    expect(screen.queryByTestId('depth-preset-select')).not.toBeInTheDocument();
  });
});
