## Test File: `src/components/Dialogs/ExportDialog.test.tsx`

```typescript
import { describe, it, expect, vi } from 'vitest';
import { render, screen, fireEvent } from '@testing-library/react';
import ExportDialog from './ExportDialog';

// Mock stores and Tauri invoke
vi.mock('../../store/uiStore', () => ({
  useUIStore: () => ({
    closeDialog: vi.fn(),
  }),
}));

vi.mock('../../store/projectStore', () => ({
  useProjectStore: () => ({
    exportCsv: vi.fn(),
    exportJson: vi.fn(),
    exportImage: vi.fn(),
  }),
}));

vi.mock('@tauri-apps/api/core', () => ({
  invoke: vi.fn().mockResolvedValue({ shallow: '#00bfff', deep: '#ff4444' }),
}));

describe('ExportDialog', () => {
  it('shows color mode options for image export', () => {
    render(<ExportDialog />);
    
    // Switch to image export
    fireEvent.click(screen.getByTestId('export-type-image'));
    
    expect(screen.getByTestId('color-mode-section')).toBeInTheDocument();
    expect(screen.getByTestId('color-mode-depth')).toBeInTheDocument();
  });

  it('shows depth preset dropdown when depth mode selected', () => {
    render(<ExportDialog />);
    
    fireEvent.click(screen.getByTestId('export-type-image'));
    fireEvent.click(screen.getByTestId('color-mode-depth'));
    
    expect(screen.getByTestId('depth-preset-select')).toBeInTheDocument();
  });

  it('shows include legend checkbox when depth mode selected', () => {
    render(<ExportDialog />);
    
    fireEvent.click(screen.getByTestId('export-type-image'));
    fireEvent.click(screen.getByTestId('color-mode-depth'));
    
    expect(screen.getByTestId('option-include-legend')).toBeInTheDocument();
  });

  it('includes 600 DPI option in resolution selector', () => {
    render(<ExportDialog />);
    
    fireEvent.click(screen.getByTestId('export-type-image'));
    
    const resolutionSelect = screen.getByTestId('resolution-select');
    expect(resolutionSelect).toContainHTML('600');
  });

  it('hides depth options when section mode selected', () => {
    render(<ExportDialog />);
    
    fireEvent.click(screen.getByTestId('export-type-image'));
    fireEvent.click(screen.getByTestId('color-mode-depth'));
    fireEvent.click(screen.getByTestId('color-mode-section'));
    
    expect(screen.queryByTestId('depth-preset-select')).not.toBeInTheDocument();
  });
});
```
