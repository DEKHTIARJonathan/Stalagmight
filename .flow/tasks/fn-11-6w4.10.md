```typescript
import { describe, it, expect } from 'vitest';
import { interpolateStops, getDepthColor, type ColorStop } from './export-colors';

// Sample preset stops (matching thermal preset)
const thermalStops: ColorStop[] = [
  { position: 0.0, color: '#00bfff' },
  { position: 1.0, color: '#ff4444' },
];

describe('interpolateStops', () => {
  it('returns first stop color for empty array', () => {
    expect(interpolateStops(0.5, [])).toBe('#000000');
  });

  it('returns single stop color for single-element array', () => {
    expect(interpolateStops(0.5, [{ position: 0, color: '#ff0000' }])).toBe('#ff0000');
  });

  it('returns shallow color at t=0', () => {
    const result = interpolateStops(0, thermalStops);
    expect(result.toLowerCase()).toBe('#00bfff');
  });

  it('returns deep color at t=1', () => {
    const result = interpolateStops(1, thermalStops);
    expect(result.toLowerCase()).toBe('#ff4444');
  });

  it('handles equal positions (avoids divide by zero)', () => {
    const stops: ColorStop[] = [
      { position: 0.5, color: '#ff0000' },
      { position: 0.5, color: '#00ff00' },
    ];
    expect(() => interpolateStops(0.5, stops)).not.toThrow();
  });
});

describe('getDepthColor', () => {
  it('returns shallow color at min depth', () => {
    const result = getDepthColor(0, 0, 100, thermalStops);
    expect(result.toLowerCase()).toBe('#00bfff');
  });

  it('returns deep color at max depth', () => {
    const result = getDepthColor(100, 0, 100, thermalStops);
    expect(result.toLowerCase()).toBe('#ff4444');
  });

  it('handles NaN depth gracefully', () => {
    const result = getDepthColor(NaN, 0, 100, thermalStops);
    expect(result).toBe('#00bfff'); // Falls back to first stop
  });

  it('handles flat depth range (min === max)', () => {
    const result = getDepthColor(50, 50, 50, thermalStops);
    expect(result).toBe('#00bfff'); // Falls back to first stop
  });

  it('clamps depth below min', () => {
    const result = getDepthColor(-10, 0, 100, thermalStops);
    expect(result.toLowerCase()).toBe('#00bfff');
  });

  it('clamps depth above max', () => {
    const result = getDepthColor(150, 0, 100, thermalStops);
    expect(result.toLowerCase()).toBe('#ff4444');
  });
});
```
