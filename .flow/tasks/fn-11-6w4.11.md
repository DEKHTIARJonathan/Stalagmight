```typescript
import { test, expect } from '@playwright/test';
import { openTestProject, openExportDialog, downloadFile } from './helpers/app';

test.describe('Export Color Modes', () => {
  test.beforeEach(async ({ page }) => {
    await openTestProject(page, 'depth-test-project.json');
    await openExportDialog(page);
  });

  test('shows color mode options for image export', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    
    await expect(page.getByTestId('color-mode-section')).toBeVisible();
    await expect(page.getByTestId('color-mode-depth')).toBeVisible();
  });

  test('section mode is default', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    
    await expect(page.getByTestId('color-mode-section')).toBeChecked();
    await expect(page.getByTestId('color-mode-depth')).not.toBeChecked();
  });

  test('depth mode shows preset dropdown', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('color-mode-depth').click();
    
    await expect(page.getByTestId('depth-preset-select')).toBeVisible();
    await expect(page.getByTestId('option-include-legend')).toBeVisible();
  });

  test('can select each depth preset', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('color-mode-depth').click();
    
    const presetSelect = page.getByTestId('depth-preset-select');
    
    for (const preset of ['thermal', 'ocean', 'terrain', 'grayscale', 'accessible']) {
      await presetSelect.selectOption(preset);
      await expect(presetSelect).toHaveValue(preset);
    }
  });

  test('preview updates when color mode changes', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    
    // Wait for initial preview
    await expect(page.getByTestId('export-preview')).toBeVisible();
    await page.waitForTimeout(500); // Debounce
    
    // Switch to depth mode
    await page.getByTestId('color-mode-depth').click();
    
    // Should show loading briefly then update
    await expect(page.getByTestId('export-preview')).toBeVisible();
  });

  test('can export PNG with depth colors', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('color-mode-depth').click();
    await page.getByTestId('depth-preset-select').selectOption('thermal');
    
    // Mock file save dialog
    const downloadPromise = page.waitForEvent('download');
    await page.getByTestId('export-button').click();
    
    const download = await downloadPromise;
    expect(download.suggestedFilename()).toMatch(/\.png$/);
  });

  test('can export SVG with depth colors', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('image-format-select').selectOption('svg');
    await page.getByTestId('color-mode-depth').click();
    await page.getByTestId('depth-preset-select').selectOption('ocean');
    
    const downloadPromise = page.waitForEvent('download');
    await page.getByTestId('export-button').click();
    
    const download = await downloadPromise;
    expect(download.suggestedFilename()).toMatch(/\.svg$/);
  });

  test('can export PDF with depth colors', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('image-format-select').selectOption('pdf');
    await page.getByTestId('color-mode-depth').click();
    
    const downloadPromise = page.waitForEvent('download');
    await page.getByTestId('export-button').click();
    
    const download = await downloadPromise;
    expect(download.suggestedFilename()).toMatch(/\.pdf$/);
  });

  test('legend checkbox controls legend in export', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('color-mode-depth').click();
    
    // Legend enabled by default
    await expect(page.getByTestId('option-include-legend')).toBeChecked();
    
    // Can toggle off
    await page.getByTestId('option-include-legend').click();
    await expect(page.getByTestId('option-include-legend')).not.toBeChecked();
  });

  test('600 DPI option available for PNG', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('image-format-select').selectOption('png');
    
    const resolutionSelect = page.getByTestId('resolution-select');
    await resolutionSelect.selectOption('600');
    await expect(resolutionSelect).toHaveValue('600');
  });

  test('shows progress during export', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('color-mode-depth').click();
    
    // Start export
    await page.getByTestId('export-button').click();
    
    // Progress should appear
    await expect(page.getByTestId('loading-progress')).toBeVisible();
    
    // Wait for completion
    await expect(page.getByTestId('loading-progress')).not.toBeVisible({ timeout: 10000 });
  });

  test('export button disabled during export', async ({ page }) => {
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('export-button').click();
    
    await expect(page.getByTestId('export-button')).toBeDisabled();
  });
});

test.describe('Export Color Modes - Edge Cases', () => {
  test('flat depth project shows "(Flat)" in legend', async ({ page }) => {
    await openTestProject(page, 'flat-depth-project.json');
    await openExportDialog(page);
    
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('image-format-select').selectOption('svg');
    await page.getByTestId('color-mode-depth').click();
    
    // Export and verify SVG content contains "(Flat)"
    const downloadPromise = page.waitForEvent('download');
    await page.getByTestId('export-button').click();
    const download = await downloadPromise;
    
    const content = await downloadFile(download);
    expect(content).toContain('(Flat)');
  });

  test('zero-datum stations have gold fill in exported SVG', async ({ page }) => {
    await openTestProject(page, 'depth-test-project.json');
    await openExportDialog(page);
    
    await page.getByTestId('export-type-image').click();
    await page.getByTestId('image-format-select').selectOption('svg');
    await page.getByTestId('color-mode-depth').click();
    
    const downloadPromise = page.waitForEvent('download');
    await page.getByTestId('export-button').click();
    const download = await downloadPromise;
    
    const content = await downloadFile(download);
    expect(content).toContain('#ffd700'); // Gold
  });
});
```
