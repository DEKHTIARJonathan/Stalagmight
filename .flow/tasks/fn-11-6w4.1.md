### HSL Interpolation (Shortest Hue Path)
```rust
/// Convert hex color to HSL
fn hex_to_hsl(hex: &str) -> (f64, f64, f64) {
    // Parse hex to RGB
    let hex = hex.trim_start_matches('#');
    let r = u8::from_str_radix(&hex[0..2], 16).unwrap_or(0) as f64 / 255.0;
    let g = u8::from_str_radix(&hex[2..4], 16).unwrap_or(0) as f64 / 255.0;
    let b = u8::from_str_radix(&hex[4..6], 16).unwrap_or(0) as f64 / 255.0;
    
    // RGB to HSL conversion
    let max = r.max(g).max(b);
    let min = r.min(g).min(b);
    let l = (max + min) / 2.0;
    
    if (max - min).abs() < f64::EPSILON {
        return (0.0, 0.0, l);
    }
    
    let d = max - min;
    let s = if l > 0.5 { d / (2.0 - max - min) } else { d / (max + min) };
    
    let h = if (max - r).abs() < f64::EPSILON {
        ((g - b) / d + if g < b { 6.0 } else { 0.0 }) / 6.0
    } else if (max - g).abs() < f64::EPSILON {
        ((b - r) / d + 2.0) / 6.0
    } else {
        ((r - g) / d + 4.0) / 6.0
    };
    
    (h * 360.0, s, l)
}

/// Interpolate between two colors using shortest hue path
pub fn get_depth_color(
    depth: f64,
    min_depth: f64,
    max_depth: f64,
    preset: DepthPreset,
) -> String {
    // Handle edge cases
    if (max_depth - min_depth).abs() < f64::EPSILON || depth.is_nan() {
        return preset.colors().shallow;
    }
    
    // Clamp and normalize
    let t = ((depth - min_depth) / (max_depth - min_depth)).clamp(0.0, 1.0);
    
    let colors = preset.colors();
    let (h1, s1, l1) = hex_to_hsl(&colors.shallow);
    let (h2, s2, l2) = hex_to_hsl(&colors.deep);
    
    // Shortest hue path interpolation
    let mut dh = h2 - h1;
    if dh > 180.0 {
        dh -= 360.0;
    } else if dh < -180.0 {
        dh += 360.0;
    }
    
    let h = (h1 + dh * t).rem_euclid(360.0);
    let s = s1 + (s2 - s1) * t;
    let l = l1 + (l2 - l1) * t;
    
    hsl_to_hex(h, s, l)
}
```
