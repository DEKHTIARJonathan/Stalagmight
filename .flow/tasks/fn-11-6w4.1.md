### Color Interpolation with NaN check AND safety guards
```rust
pub fn get_depth_color(depth: f64, min_depth: f64, max_depth: f64, preset: DepthPreset) -> String {
    // CRITICAL: Check for NaN/infinite values
    if !depth.is_finite() || !min_depth.is_finite() || !max_depth.is_finite() {
        return preset.colors().stops[0].color.clone();
    }
    
    if (max_depth - min_depth).abs() < f64::EPSILON {
        return preset.colors().stops[0].color.clone();
    }
    
    let t = ((depth - min_depth) / (max_depth - min_depth)).clamp(0.0, 1.0);
    interpolate_stops(t, &preset.colors().stops)
}

fn interpolate_stops(t: f64, stops: &[ColorStop]) -> String {
    // CRITICAL: Safety guards for edge cases
    if stops.is_empty() {
        return "#000000".to_string();  // Fallback black
    }
    if stops.len() == 1 {
        return stops[0].color.clone();
    }
    
    let t = t.clamp(0.0, 1.0);
    
    for i in 0..stops.len() - 1 {
        if t >= stops[i].position && t <= stops[i + 1].position {
            // Handle equal positions (avoid divide by zero)
            if (stops[i + 1].position - stops[i].position).abs() < f64::EPSILON {
                return stops[i].color.clone();
            }
            let local_t = (t - stops[i].position) / (stops[i + 1].position - stops[i].position);
            return interpolate_rgb(&stops[i].color, &stops[i + 1].color, local_t);
        }
    }
    stops.last().unwrap().color.clone()
}
```

### Multi-Stop Interpolation for Thermal
```rust
const THERMAL_STOPS: &[(f64, &str)] = &[
    (0.0, "#00bfff"),  // Blue (shallow)
    (0.25, "#00ffaa"), // Cyan-green
    (0.5, "#44ff00"),  // Green
    (0.75, "#ffff00"), // Yellow
    (1.0, "#ff4444"),  // Red (deep)
];

fn interpolate_multi_stop(t: f64, stops: &[(f64, &str)]) -> String {
    // Find the two stops that bracket t
    for i in 0..stops.len() - 1 {
        if t >= stops[i].0 && t <= stops[i + 1].0 {
            let local_t = (t - stops[i].0) / (stops[i + 1].0 - stops[i].0);
            return interpolate_rgb(stops[i].1, stops[i + 1].1, local_t);
        }
    }
    stops.last().unwrap().1.to_string()
}
```

```rust
pub fn get_depth_color(depth: f64, min_depth: f64, max_depth: f64, preset: DepthPreset) -> String {
    if depth.is_nan() || (max_depth - min_depth).abs() < f64::EPSILON {
```

### RGB Interpolation (simpler than HSL for two-point)
```rust
fn interpolate_rgb(color1: &str, color2: &str, t: f64) -> String {
    let (r1, g1, b1) = hex_to_rgb(color1);
    let (r2, g2, b2) = hex_to_rgb(color2);
    
    let r = (r1 as f64 + (r2 as f64 - r1 as f64) * t) as u8;
    let g = (g1 as f64 + (g2 as f64 - g1 as f64) * t) as u8;
    let b = (b1 as f64 + (b2 as f64 - b1 as f64) * t) as u8;
    
    format!("#{:02x}{:02x}{:02x}", r, g, b)
}
```

### HSL Interpolation (Shortest Hue Path)
```rust
/// Convert hex color to HSL
fn hex_to_hsl(hex: &str) -> (f64, f64, f64) {
    // Parse hex to RGB
    let hex = hex.trim_start_matches('#');
    let r = u8::from_str_radix(&hex[0..2], 16).unwrap_or(0) as f64 / 255.0;
    let g = u8::from_str_radix(&hex[2..4], 16).unwrap_or(0) as f64 / 255.0;
    let b = u8::from_str_radix(&hex[4..6], 16).unwrap_or(0) as f64 / 255.0;
    
    // RGB to HSL conversion
    let max = r.max(g).max(b);
    let min = r.min(g).min(b);
    let l = (max + min) / 2.0;
    
    if (max - min).abs() < f64::EPSILON {
        return (0.0, 0.0, l);
    }
    
    let d = max - min;
    let s = if l > 0.5 { d / (2.0 - max - min) } else { d / (max + min) };
    
    let h = if (max - r).abs() < f64::EPSILON {
        ((g - b) / d + if g < b { 6.0 } else { 0.0 }) / 6.0
    } else if (max - g).abs() < f64::EPSILON {
        ((b - r) / d + 2.0) / 6.0
    } else {
        ((r - g) / d + 4.0) / 6.0
    };
    
    (h * 360.0, s, l)
}

/// Interpolate between two colors using shortest hue path
pub fn get_depth_color(
    depth: f64,
    min_depth: f64,
    max_depth: f64,
    preset: DepthPreset,
) -> String {
    // Handle edge cases
    if (max_depth - min_depth).abs() < f64::EPSILON || depth.is_nan() {
        return preset.colors().shallow;
    }
    
    // Clamp and normalize
    let t = ((depth - min_depth) / (max_depth - min_depth)).clamp(0.0, 1.0);
    
    let colors = preset.colors();
    let (h1, s1, l1) = hex_to_hsl(&colors.shallow);
    let (h2, s2, l2) = hex_to_hsl(&colors.deep);
    
    // Shortest hue path interpolation
    let mut dh = h2 - h1;
    if dh > 180.0 {
        dh -= 360.0;
    } else if dh < -180.0 {
        dh += 360.0;
    }
    
    let h = (h1 + dh * t).rem_euclid(360.0);
    let s = s1 + (s2 - s1) * t;
    let l = l1 + (l2 - l1) * t;
    
    hsl_to_hex(h, s, l)
}
```
