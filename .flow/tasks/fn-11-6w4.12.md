# fn-11-6w4.12 Write WebdriverIO accessibility tests for export dialog

## Description
Write WebdriverIO accessibility tests for the export dialog color mode features.

**Depends on:** fn-11-6w4.11 (Playwright E2E tests)

## Test File: `e2e/specs/export-accessibility.spec.ts`

```typescript
import { browser, $ } from '@wdio/globals';

describe('Export Dialog Accessibility', () => {
  beforeEach(async () => {
    // Open test project and export dialog
    await browser.execute(() => window.testHelpers.openProject('depth-test-project.json'));
    await browser.execute(() => window.testHelpers.openExportDialog());
  });

  describe('Color Mode Selection', () => {
    it('should have accessible radio group for color mode', async () => {
      await $('[data-testid="export-type-image"]').click();
      
      const sectionRadio = await $('[data-testid="color-mode-section"]');
      const depthRadio = await $('[data-testid="color-mode-depth"]');
      
      // Both radios should have same name (grouped)
      expect(await sectionRadio.getAttribute('name')).toBe(await depthRadio.getAttribute('name'));
      
      // Should be keyboard accessible
      await sectionRadio.click();
      await browser.keys('Tab');
      expect(await depthRadio.isFocused()).toBe(true);
    });

    it('should announce color mode change to screen readers', async () => {
      await $('[data-testid="export-type-image"]').click();
      await $('[data-testid="color-mode-depth"]').click();
      
      // Verify ARIA attributes
      const depthRadio = await $('[data-testid="color-mode-depth"]');
      expect(await depthRadio.getAttribute('aria-checked')).toBe('true');
    });
  });

  describe('Depth Preset Dropdown', () => {
    beforeEach(async () => {
      await $('[data-testid="export-type-image"]').click();
      await $('[data-testid="color-mode-depth"]').click();
    });

    it('should have accessible label for preset dropdown', async () => {
      const select = await $('[data-testid="depth-preset-select"]');
      const labelId = await select.getAttribute('aria-labelledby');
      
      if (labelId) {
        const label = await $(`#${labelId}`);
        expect(await label.getText()).toContain('Color Preset');
      } else {
        // Check for label element
        const label = await select.$('xpath=./ancestor::div//label');
        expect(await label.getText()).toContain('Color Preset');
      }
    });

    it('should be navigable with keyboard', async () => {
      const select = await $('[data-testid="depth-preset-select"]');
      await select.click();
      
      // Navigate through options
      await browser.keys('ArrowDown');
      await browser.keys('ArrowDown');
      await browser.keys('Enter');
      
      // Should have selected a different option
      expect(await select.getValue()).not.toBe('thermal');
    });

    it('should describe preset options clearly', async () => {
      const select = await $('[data-testid="depth-preset-select"]');
      const options = await select.$$('option');
      
      // Each option should have descriptive text
      for (const option of options) {
        const text = await option.getText();
        expect(text.length).toBeGreaterThan(5);
        expect(text).toMatch(/\(.*\)/); // Should have parenthetical description
      }
    });
  });

  describe('Legend Checkbox', () => {
    beforeEach(async () => {
      await $('[data-testid="export-type-image"]').click();
      await $('[data-testid="color-mode-depth"]').click();
    });

    it('should have accessible checkbox with label', async () => {
      const checkbox = await $('[data-testid="option-include-legend"]');
      const label = await checkbox.$('xpath=./ancestor::label');
      
      expect(await label.getText()).toContain('legend');
    });

    it('should be toggleable with keyboard', async () => {
      const checkbox = await $('[data-testid="option-include-legend"]');
      await checkbox.click();
      
      const wasChecked = await checkbox.isSelected();
      await browser.keys('Space');
      expect(await checkbox.isSelected()).toBe(!wasChecked);
    });
  });

  describe('Preview Accessibility', () => {
    it('should have accessible preview with alt text', async () => {
      await $('[data-testid="export-type-image"]').click();
      
      const preview = await $('[data-testid="export-preview"]');
      
      // Should have role or be a recognizable element
      const role = await preview.getAttribute('role');
      expect(['img', 'figure', null]).toContain(role);
      
      // Should have accessible name or aria-label
      const ariaLabel = await preview.getAttribute('aria-label');
      if (ariaLabel) {
        expect(ariaLabel).toContain('preview');
      }
    });

    it('should announce loading state', async () => {
      await $('[data-testid="export-type-image"]').click();
      await $('[data-testid="color-mode-depth"]').click();
      
      // Change preset to trigger reload
      const select = await $('[data-testid="depth-preset-select"]');
      await select.selectByVisibleText('Ocean (Cyan → Blue)');
      
      const loading = await $('[data-testid="export-preview-loading"]');
      const ariaLive = await loading.getAttribute('aria-live');
      expect(['polite', 'assertive']).toContain(ariaLive);
    });
  });

  describe('Export Button', () => {
    it('should announce disabled state when exporting', async () => {
      await $('[data-testid="export-type-image"]').click();
      await $('[data-testid="export-button"]').click();
      
      const button = await $('[data-testid="export-button"]');
      expect(await button.getAttribute('aria-disabled')).toBe('true');
    });

    it('should have clear button text indicating action', async () => {
      await $('[data-testid="export-type-image"]').click();
      
      const button = await $('[data-testid="export-button"]');
      expect(await button.getText()).toBe('Export');
    });
  });

  describe('Resolution Select', () => {
    it('should explain DPI options accessibly', async () => {
      await $('[data-testid="export-type-image"]').click();
      
      const select = await $('[data-testid="resolution-select"]');
      const options = await select.$$('option');
      
      // Should have descriptive labels including 600 DPI
      const optionTexts = await Promise.all(options.map(o => o.getText()));
      expect(optionTexts.some(t => t.includes('600'))).toBe(true);
      expect(optionTexts.some(t => t.includes('Print'))).toBe(true);
    });
  });
});
```
