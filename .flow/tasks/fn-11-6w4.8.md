## Backend Progress Events

Emit progress events during export in `src-tauri/src/commands/export.rs`:

```rust
use crate::events::LoadingProgress;

#[tauri::command]
pub async fn export_image(
    state: State<'_, AppState>,
    app_handle: tauri::AppHandle,
    path: PathBuf,
    options: ExportImageOptions,
) -> Result<(), String> {
    let project = state
        .get_project_state()
        .ok_or_else(|| "No project open".to_string())?;
    
    // Start progress
    let _ = app_handle.emit("loading:progress", LoadingProgress {
        stage: "Preparing export".to_string(),
        percent: 10,
        message: None,
    });
    
    // Generate SVG
    let _ = app_handle.emit("loading:progress", LoadingProgress {
        stage: "Generating SVG".to_string(),
        percent: 30,
        message: Some(format!("{} stations", project.compute_station_coordinates().len())),
    });
    
    // For PNG, additional rasterization step
    if options.format == ImageFormat::Png {
        let _ = app_handle.emit("loading:progress", LoadingProgress {
            stage: "Rasterizing PNG".to_string(),
            percent: 60,
            message: Some(format!("{} DPI", options.dpi)),
        });
    }
    
    // Actual export
    let result = crate::export::export_image(&project, &path, &options);
    
    // Complete
    let _ = app_handle.emit("loading:progress", LoadingProgress {
        stage: "Export complete".to_string(),
        percent: 100,
        message: None,
    });
    
    result.map_err(|e| e.to_string())
}
```

import { useLoadingProgress } from '../../hooks/useLoadingProgress';
import { LoadingProgress } from '../common/LoadingProgress';

export default function ExportDialog() {
  const [isExporting, setIsExporting] = useState(false);
  const loadingProgress = useLoadingProgress('export');
  
  const handleExport = async () => {
    setIsExporting(true);
    try {
      // ... existing export logic ...
    } finally {
      setIsExporting(false);
    }
  };
  
  return (
    <DialogWrapper>
      {/* ... existing content ... */}
      
      {isExporting && loadingProgress && (
        <div className="mt-4">
          <LoadingProgress 
            stage={loadingProgress.stage}
            percent={loadingProgress.percent}
            message={loadingProgress.message}
          />
        </div>
      )}
      
      {/* Actions - disable during export */}
      <div className="flex justify-end gap-2 pt-4 mt-4 border-t border-border">
        <Button onClick={closeDialog} disabled={isExporting}>Cancel</Button>
        <Button 
          variant="primary" 
          onClick={handleExport} 
          disabled={isExporting}
          data-testid="export-button"
        >
          {isExporting ? 'Exporting...' : 'Export'}
        </Button>
      </div>
    </DialogWrapper>
  );
}
