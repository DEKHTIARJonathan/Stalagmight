export interface ColorStop {
  position: number;
  color: string;
}

export interface DepthRange {
  min: number;
  max: number;
}

/**
 * Parse hex color to RGB components.
 */
function hexToRgb(hex: string): { r: number; g: number; b: number } {
  const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
  return result ? {
    r: parseInt(result[1], 16),
    g: parseInt(result[2], 16),
    b: parseInt(result[3], 16),
  } : { r: 0, g: 0, b: 0 };
}

/**
 * Interpolate between two hex colors.
 */
function interpolateRgb(color1: string, color2: string, t: number): string {
  const c1 = hexToRgb(color1);
  const c2 = hexToRgb(color2);
  const r = Math.round(c1.r + (c2.r - c1.r) * t);
  const g = Math.round(c1.g + (c2.g - c1.g) * t);
  const b = Math.round(c1.b + (c2.b - c1.b) * t);
  return `#${r.toString(16).padStart(2, '0')}${g.toString(16).padStart(2, '0')}${b.toString(16).padStart(2, '0')}`;
}

/**
 * Multi-stop interpolation matching Rust backend.
 * CRITICAL: Same safety guards as Rust implementation.
 */
export function interpolateStops(t: number, stops: ColorStop[]): string {
  // Guard for empty array (match Rust backend)
  if (stops.length === 0) {
    return '#000000';
  }
  
  // Guard for single-element array
  if (stops.length === 1) {
    return stops[0].color;
  }
  
  // Clamp t to valid range
  const clampedT = Math.max(0, Math.min(1, t));
  
  // Find bracketing stops (assumes sorted by position)
  for (let i = 0; i < stops.length - 1; i++) {
    if (clampedT >= stops[i].position && clampedT <= stops[i + 1].position) {
      const denominator = stops[i + 1].position - stops[i].position;
      // Guard for equal positions (avoid divide by zero)
      if (Math.abs(denominator) < Number.EPSILON) {
        return stops[i].color;
      }
      const localT = (clampedT - stops[i].position) / denominator;
      return interpolateRgb(stops[i].color, stops[i + 1].color, localT);
    }
  }
  
  return stops[stops.length - 1].color;
}

/**
 * Compute depth color from multi-stop preset.
 */
export function getDepthColorMultiStop(
  depth: number,
  minDepth: number,
  maxDepth: number,
  stops: ColorStop[],
): string {
  // Handle NaN/Infinite
  if (!Number.isFinite(depth) || !Number.isFinite(minDepth) || !Number.isFinite(maxDepth)) {
    return stops.length > 0 ? stops[0].color : '#000000';
  }
  
  // Handle flat range
  if (Math.abs(maxDepth - minDepth) < Number.EPSILON) {
    return stops.length > 0 ? stops[0].color : '#000000';
  }
  
  const t = Math.max(0, Math.min(1, (depth - minDepth) / (maxDepth - minDepth)));
  return interpolateStops(t, stops);
}

/**
 * Compute station color based on color mode.
 */
export function computeStationColor(
  station: { coordinates: { z: number }; is_zero_datum: boolean },
  sectionColor: string,
  colorMode: 'section' | 'depth',
  depthRange: DepthRange,
  presetStops: ColorStop[] | null,
): string {
  // Zero datum always gold
  if (station.is_zero_datum) {
    return '#ffd700';
  }
  
  // Section mode or no preset
  if (colorMode === 'section' || !presetStops || presetStops.length === 0) {
    return sectionColor;
  }
  
  // Depth mode with multi-stop interpolation
  return getDepthColorMultiStop(
    station.coordinates.z,
    depthRange.min,
    depthRange.max,
    presetStops,
  );
}

/**
 * Compute shot color based on color mode (uses average depth).
 */
export function computeShotColor(
  fromZ: number,
  toZ: number,
  sectionColor: string,
  colorMode: 'section' | 'depth',
  depthRange: DepthRange,
  presetStops: ColorStop[] | null,
): string {
  if (colorMode === 'section' || !presetStops || presetStops.length === 0) {
    return sectionColor;
  }
  
  const avgDepth = (fromZ + toZ) / 2;
  return getDepthColorMultiStop(avgDepth, depthRange.min, depthRange.max, presetStops);
}

/**
 * Calculate depth range from stations.
 */
export function calculateDepthRange(stations: Array<{ coordinates: { z: number } }>): DepthRange {
  if (stations.length === 0) {
    return { min: 0, max: 50 };
  }
  
  const depths = stations.map(s => s.coordinates.z);
  return {
    min: Math.min(...depths),
    max: Math.max(...depths),
  };
}

  min: number;
  max: number;

  deep: string;
