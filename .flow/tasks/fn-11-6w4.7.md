## New File: `src/utils/export-colors.ts`
```typescript
import { getDepthColor } from './colors';
import type { Station } from '../types';

export interface DepthRange {
  min: number;
  max: number;
}

export interface PresetColors {
  shallow: string;
  deep: string;
}

/**
 * Compute station color based on color mode.
 * Pure function - easy to unit test.
 */
export function computeStationColor(
  station: { coordinates: { z: number }; is_zero_datum: boolean },
  sectionColor: string,
  colorMode: 'section' | 'depth',
  depthRange: DepthRange,
  presetColors: PresetColors | null,
): string {
  // Zero datum always gold
  if (station.is_zero_datum) {
    return '#ffd700';
  }
  
  // Section mode
  if (colorMode === 'section' || !presetColors) {
    return sectionColor;
  }
  
  // Depth mode
  return getDepthColor(
    station.coordinates.z,
    presetColors.shallow,
    presetColors.deep,
    depthRange.min,
    depthRange.max
  ) ?? sectionColor;
}

/**
 * Compute shot color based on color mode.
 */
export function computeShotColor(
  fromZ: number,
  toZ: number,
  sectionColor: string,
  colorMode: 'section' | 'depth',
  depthRange: DepthRange,
  presetColors: PresetColors | null,
): string {
  if (colorMode === 'section' || !presetColors) {
    return sectionColor;
  }
  
  const avgDepth = (fromZ + toZ) / 2;
  return getDepthColor(
    avgDepth,
    presetColors.shallow,
    presetColors.deep,
    depthRange.min,
    depthRange.max
  ) ?? sectionColor;
}

/**
 * Calculate depth range from stations.
 */
export function calculateDepthRange(stations: Array<{ coordinates: { z: number } }>): DepthRange {
  if (stations.length === 0) {
    return { min: 0, max: 50 };
  }
  
  const depths = stations.map(s => s.coordinates.z);
  return {
    min: Math.min(...depths),
    max: Math.max(...depths),
  };
}
```
