```rust
#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_preset_colors() {
        assert_eq!(DepthPreset::Thermal.colors().shallow, "#00bfff");
        assert_eq!(DepthPreset::Thermal.colors().deep, "#ff4444");
        
        assert_eq!(DepthPreset::Ocean.colors().shallow, "#00ffff");
        assert_eq!(DepthPreset::Ocean.colors().deep, "#0000aa");
        
        assert_eq!(DepthPreset::Terrain.colors().shallow, "#00ff00");
        assert_eq!(DepthPreset::Terrain.colors().deep, "#8b4513");
        
        assert_eq!(DepthPreset::Grayscale.colors().shallow, "#ffffff");
        assert_eq!(DepthPreset::Grayscale.colors().deep, "#000000");
        
        assert_eq!(DepthPreset::Accessible.colors().shallow, "#fde047");
        assert_eq!(DepthPreset::Accessible.colors().deep, "#7c3aed");
    }

    #[test]
    fn test_hex_to_hsl_basic() {
        // Pure red
        let (h, s, l) = hex_to_hsl("#ff0000");
        assert!((h - 0.0).abs() < 1.0);
        assert!((s - 1.0).abs() < 0.01);
        assert!((l - 0.5).abs() < 0.01);
        
        // Pure blue
        let (h, s, l) = hex_to_hsl("#0000ff");
        assert!((h - 240.0).abs() < 1.0);
        
        // White
        let (h, s, l) = hex_to_hsl("#ffffff");
        assert!((l - 1.0).abs() < 0.01);
        
        // Black
        let (h, s, l) = hex_to_hsl("#000000");
        assert!((l - 0.0).abs() < 0.01);
    }

    #[test]
    fn test_hsl_to_hex_roundtrip() {
        let colors = ["#ff0000", "#00ff00", "#0000ff", "#ffffff", "#000000", "#808080"];
        for color in colors {
            let (h, s, l) = hex_to_hsl(color);
            let result = hsl_to_hex(h, s, l);
            assert_eq!(result.to_lowercase(), color.to_lowercase());
        }
    }

    #[test]
    fn test_get_depth_color_min_max() {
        // At min depth, should return shallow color
        let shallow = get_depth_color(0.0, 0.0, 100.0, DepthPreset::Thermal);
        assert_eq!(shallow, "#00bfff");
        
        // At max depth, should return deep color
        let deep = get_depth_color(100.0, 0.0, 100.0, DepthPreset::Thermal);
        assert_eq!(deep, "#ff4444");
    }

    #[test]
    fn test_get_depth_color_midpoint() {
        // At midpoint, should be somewhere between
        let mid = get_depth_color(50.0, 0.0, 100.0, DepthPreset::Thermal);
        // Should be greenish (between blue and red via shortest hue path)
        assert!(mid.starts_with("#"));
        assert_ne!(mid, "#00bfff");
        assert_ne!(mid, "#ff4444");
    }

    #[test]
    fn test_get_depth_color_clamp() {
        // Below min should clamp to shallow
        let below = get_depth_color(-10.0, 0.0, 100.0, DepthPreset::Thermal);
        assert_eq!(below, "#00bfff");
        
        // Above max should clamp to deep
        let above = get_depth_color(150.0, 0.0, 100.0, DepthPreset::Thermal);
        assert_eq!(above, "#ff4444");
    }

    #[test]
    fn test_get_depth_color_same_min_max() {
        // When min == max, should return shallow
        let result = get_depth_color(50.0, 50.0, 50.0, DepthPreset::Thermal);
        assert_eq!(result, "#00bfff");
    }

    #[test]
    fn test_get_depth_color_nan() {
        // NaN depth should return shallow
        let result = get_depth_color(f64::NAN, 0.0, 100.0, DepthPreset::Thermal);
        assert_eq!(result, "#00bfff");
    }

    #[test]
    fn test_shortest_hue_path() {
        // Thermal goes blue (195°) → red (0°/360°)
        // Shortest path is through cyan → green → yellow → red
        // NOT through purple
        let quarter = get_depth_color(25.0, 0.0, 100.0, DepthPreset::Thermal);
        let half = get_depth_color(50.0, 0.0, 100.0, DepthPreset::Thermal);
        let three_quarter = get_depth_color(75.0, 0.0, 100.0, DepthPreset::Thermal);
        
        // Parse hues and verify they go through green/yellow range
        let (h1, _, _) = hex_to_hsl(&quarter);
        let (h2, _, _) = hex_to_hsl(&half);
        let (h3, _, _) = hex_to_hsl(&three_quarter);
        
        // Hues should progress from blue (~195) toward red (0/360)
        // via green (~120) range
        assert!(h1 > 90.0 && h1 < 200.0, "Quarter should be cyan/green range");
        assert!(h2 > 60.0 && h2 < 150.0, "Half should be green/yellow range");
        assert!(h3 < 90.0 || h3 > 300.0, "Three-quarter should be yellow/red range");
    }

    #[test]
    fn test_all_presets_produce_valid_colors() {
        let presets = [
            DepthPreset::Thermal,
            DepthPreset::Ocean,
            DepthPreset::Terrain,
            DepthPreset::Grayscale,
            DepthPreset::Accessible,
        ];
        
        for preset in presets {
            for depth in [0.0, 25.0, 50.0, 75.0, 100.0] {
                let color = get_depth_color(depth, 0.0, 100.0, preset);
                assert!(color.starts_with("#"), "Should be hex color");
                assert_eq!(color.len(), 7, "Should be #RRGGBB format");
            }
        }
    }
}
```
