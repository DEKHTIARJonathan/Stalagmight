## Update PNG Export

Modify `src-tauri/src/export/image.rs`:

```rust
use resvg;
use tiny_skia;

pub fn export_image(project: &Project, path: &Path, options: &ExportImageOptions) -> Result<()> {
    match options.format {
        ImageFormat::Svg => export_svg(project, path, options),
        ImageFormat::Pdf => export_pdf(project, path, options),
        ImageFormat::Png => export_png(project, path, options),  // NEW
    }
}

fn export_png(project: &Project, path: &Path, options: &ExportImageOptions) -> Result<()> {
    // Generate SVG content first
    let svg_content = generate_svg_string(project, options)?;
    
    // Parse SVG
    let opt = usvg::Options::default();
    let tree = usvg::Tree::from_str(&svg_content, &opt)
        .context("Failed to parse SVG for PNG conversion")?;
    
    // Calculate size with DPI scaling
    let size = tree.size();
    let scale = options.dpi as f32 / 72.0;
    let width = (size.width() * scale) as u32;
    let height = (size.height() * scale) as u32;
    
    // Validate size (prevent OOM)
    if width > 16384 || height > 16384 {
        anyhow::bail!("Export size too large. Reduce DPI or project extent.");
    }
    
    // Create pixmap
    let mut pixmap = tiny_skia::Pixmap::new(width, height)
        .ok_or_else(|| anyhow::anyhow!("Failed to create pixmap"))?;
    
    // Render SVG to pixmap
    let transform = tiny_skia::Transform::from_scale(scale, scale);
    resvg::render(&tree, transform, &mut pixmap.as_mut());
    
    // Save PNG
    pixmap.save_png(path)
        .context("Failed to save PNG file")?;
    
    tracing::info!(path = ?path, dpi = options.dpi, "Exported PNG");
    Ok(())
}
```
