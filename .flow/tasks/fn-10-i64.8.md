Add error state rendering:

```tsx
export function LoadingProgress({
  // ... existing props
  error,
  onRetry,
}: LoadingProgressProps & { onRetry?: () => void }) {
  const [detailsExpanded, setDetailsExpanded] = React.useState(false);

  const handleCopyError = () => {
    if (error) {
      const errorText = `${error.message}${error.details ? '\n\n' + error.details : ''}`;
      navigator.clipboard.writeText(errorText);
    }
  };

  if (error) {
    return (
      <div 
        className="w-80 bg-surface rounded-lg shadow-lg p-6"
        role="alert"
        aria-live="assertive"
      >
        <h2 className="text-lg font-medium text-error mb-2">
          Failed to Load Project
        </h2>
        
        <p className="text-text-secondary mb-3">
          {error.message}
        </p>

        {error.details && (
          <div className="mb-4">
            <button
              onClick={() => setDetailsExpanded(!detailsExpanded)}
              className="text-sm text-text-tertiary hover:text-text-secondary"
            >
              {detailsExpanded ? '▼ Hide details' : '▶ Show details'}
            </button>
            {detailsExpanded && (
              <pre className="mt-2 p-2 bg-surface-alt rounded text-xs text-text-tertiary overflow-auto max-h-32">
                {error.details}
              </pre>
            )}
          </div>
        )}

        <div className="flex gap-2">
          {onRetry && (
            <button
              onClick={onRetry}
              className="px-3 py-1.5 bg-accent text-white rounded hover:bg-accent-hover"
            >
              Retry
            </button>
          )}
          <button
            onClick={onCancel}
            className="px-3 py-1.5 bg-surface-alt text-text-primary rounded hover:bg-surface-hover"
          >
            Dismiss
          </button>
          <button
            onClick={handleCopyError}
            className="px-3 py-1.5 text-text-secondary hover:text-text-primary"
            title="Copy error to clipboard"
          >
            Copy
          </button>
        </div>
      </div>
    );
  }

  // ... existing progress rendering
}
```
