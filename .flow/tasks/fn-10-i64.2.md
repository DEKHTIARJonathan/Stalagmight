## Implementation in state.rs

```rust
impl AppState {
    pub fn do_load_project_full(&self, path: PathBuf) -> Result<LoadProjectFullResult, String> {
        let handle = self.get_app_handle().ok_or("AppHandle not initialized")?;
        let token = &self.loading_cancellation;
        token.reset();
        
        let mut progress = ProgressEmitter::new_for_open(handle);

        // Check already open
        if let Some(current) = self.project.lock().unwrap().as_ref() {
            if current.file_path == Some(path.clone()) {
                return Err("ALREADY_OPEN".to_string());
            }
        }

        // Phase 0: Opening database (5%)
        progress.start_phase(LoadingPhase::OpeningDatabase);
        if !path.exists() {
            return Err(format!("File not found: {:?}", path));
        }
        let db = Database::open_existing(&path).map_err(|e| e.to_string())?;
        if token.is_cancelled() { return Err("Cancelled".to_string()); }
        progress.complete_phase(LoadingPhase::OpeningDatabase);

        // Phase 1: Loading project data (20%)
        progress.start_phase(LoadingPhase::LoadingProjectData);
        let mut project = db.load_project().map_err(|e| e.to_string())?;
        progress.update_phase(LoadingPhase::LoadingProjectData, 1, 1);
        if token.is_cancelled() { return Err("Cancelled".to_string()); }
        progress.complete_phase(LoadingPhase::LoadingProjectData);

        // Phase 2: Loading sections (15%)
        progress.start_phase(LoadingPhase::LoadingSections);
        let sections = project.sections();
        let total_sections = sections.len() as u32;
        for (i, _section) in sections.iter().enumerate() {
            if token.is_cancelled() { return Err("Cancelled".to_string()); }
            progress.update_phase(LoadingPhase::LoadingSections, i as u32 + 1, total_sections);
        }
        progress.complete_phase(LoadingPhase::LoadingSections);

        // Phase 3: Computing stations (20%)
        progress.start_phase(LoadingPhase::ComputingStations);
        let stations = self.compute_stations_with_progress(&project, &mut progress, token)?;
        progress.complete_phase(LoadingPhase::ComputingStations);

        // Phase 4: Computing statistics (15%)
        progress.start_phase(LoadingPhase::ComputingStatistics);
        let stats = self.compute_stats_with_progress(&project, &mut progress, token)?;
        progress.complete_phase(LoadingPhase::ComputingStatistics);

        // Phase 5: Validating data (10%)
        progress.start_phase(LoadingPhase::ValidatingData);
        let warnings = self.validate_project(&project);
        if token.is_cancelled() { return Err("Cancelled".to_string()); }
        progress.complete_phase(LoadingPhase::ValidatingData);

        // Phase 6: Loading settings (5%)
        progress.start_phase(LoadingPhase::LoadingSettings);
        let display_settings = db.load_display_settings().unwrap_or_default();
        progress.complete_phase(LoadingPhase::LoadingSettings);

        // Phase 7: Finalizing (10%)
        progress.start_phase(LoadingPhase::Finalizing);
        // Store project, update recent
        *self.project.lock().unwrap() = Some(project.clone());
        self.add_to_recent(&path)?;
        progress.complete_phase(LoadingPhase::Finalizing);

        Ok(LoadProjectFullResult {
            loading_id: progress.loading_id(),
            project,
            stations,
            stats,
            warnings,
            display_settings,
        })
    }

    fn compute_stations_with_progress(
        &self,
        project: &Project,
        progress: &mut ProgressEmitter,
        token: &CancellationToken,
    ) -> Result<Vec<ComputedStation>, String> {
        let shots: Vec<_> = project.all_shots().collect();
        let total = (shots.len() * 2) as u32; // approximate station count
        let mut stations = Vec::new();
        
        for (i, shot) in shots.iter().enumerate() {
            if token.is_cancelled() { return Err("Cancelled".to_string()); }
            // Compute from and to stations
            let computed = compute_station_coords(shot);
            stations.extend(computed);
            progress.update_phase(LoadingPhase::ComputingStations, i as u32 + 1, total / 2);
        }
        
        Ok(stations)
    }
}
```

## Why New Command?

The full load pipeline currently spans multiple commands:
- `open_project` - opens DB, loads project data
- `get_project_state` - computes stations, stats  
- `get_project_warnings` - validates data
- `load_display_settings` - UI settings

Creating a single `load_project_full` command allows linear progress emission from one synchronous flow.

```rust
use crate::core::loading::{LoadingPhase, ProgressEmitter, CancellationToken};

fn load_project_full(&self, path: PathBuf) -> CommandResult {
    let handle = self.get_app_handle()
        .ok_or_else(|| CommandResult::failure("AppHandle not initialized"))?;
    
    let token = &self.cancellation_token;
    token.reset();
    
    let mut progress = ProgressEmitter::new_for_open(handle);

    // Phase 0: Opening database (5%)
    progress.start_phase(LoadingPhase::OpeningDatabase);
    if !path.exists() {
        return CommandResult::failure(format!("File not found: {:?}", path));
    }
    let db = match Database::open_existing(&path) { ... };
    if token.is_cancelled() { return CommandResult::failure("Cancelled"); }
    progress.complete_phase(LoadingPhase::OpeningDatabase);

    // Phase 1: Loading project data (20%)
    progress.start_phase(LoadingPhase::LoadingProjectData);
    let mut project = match db.load_project() { ... };
    // Granular progress during load if possible
    if token.is_cancelled() { return CommandResult::failure("Cancelled"); }
    progress.complete_phase(LoadingPhase::LoadingProjectData);

    // Phase 2: Loading sections (15%)
    progress.start_phase(LoadingPhase::LoadingSections);
    let sections = project.sections();
    let total_sections = sections.len() as u32;
    for (i, _section) in sections.iter().enumerate() {
        progress.update_phase(LoadingPhase::LoadingSections, i as u32 + 1, total_sections);
        if token.is_cancelled() { return CommandResult::failure("Cancelled"); }
    }
    progress.complete_phase(LoadingPhase::LoadingSections);

    // Phase 3: Computing stations (20%)
    progress.start_phase(LoadingPhase::ComputingStations);
    // Call project.compute_station_coordinates() with progress callbacks
    // Or refactor to iterate with progress updates
    progress.complete_phase(LoadingPhase::ComputingStations);

    // Phase 4: Computing statistics (15%)
    progress.start_phase(LoadingPhase::ComputingStatistics);
    // Compute stats with progress
    progress.complete_phase(LoadingPhase::ComputingStatistics);

    // Phase 5: Validating data (10%)
    progress.start_phase(LoadingPhase::ValidatingData);
    // Run validation (heartbeat during long validation)
    progress.complete_phase(LoadingPhase::ValidatingData);

    // Phase 6: Loading settings (5%)
    progress.start_phase(LoadingPhase::LoadingSettings);
    // Load display settings
    progress.complete_phase(LoadingPhase::LoadingSettings);

    // Phase 7: Finalizing (10%)
    progress.start_phase(LoadingPhase::Finalizing);
    // Store project, update recent, etc.
    progress.complete_phase(LoadingPhase::Finalizing);

    CommandResult::success_with_data(LoadProjectResult {
        loading_id: progress.loading_id(),
        // ... project state data
    })
}
```

```rust
impl AppState {

        *self.app_handle.lock().unwrap() = Some(handle);
    }

    fn open_project(&self, path: PathBuf) -> CommandResult {
        let handle = self.app_handle.lock().unwrap().clone()
            .ok_or_else(|| CommandResult::failure("AppHandle not initialized"))?;
        
        let mut progress = ProgressEmitter::new(handle);
        let token = self.get_cancellation_token();

        // Phase 1: Opening database
        progress.start_phase(LoadingPhase::OpeningDatabase);
        if !path.exists() {
            return CommandResult::failure(format!("File not found: {:?}", path));
        }
        let db = match Database::open_existing(&path) {
            Ok(db) => db,
            Err(e) => return CommandResult::failure(format!("Failed to open database: {}", e)),
        };
        if token.is_cancelled() { return CommandResult::failure("Cancelled"); }
        progress.complete_phase(LoadingPhase::OpeningDatabase);

        // Phase 2: Loading project data
        progress.start_phase(LoadingPhase::LoadingProjectData);
        let mut project = match db.load_project() {
            Ok(p) => p,
            Err(e) => return CommandResult::failure(format!("Failed to load project: {}", e)),
        };
        if token.is_cancelled() { return CommandResult::failure("Cancelled"); }
```
