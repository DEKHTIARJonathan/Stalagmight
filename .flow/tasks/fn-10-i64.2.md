## Implementation with ATOMIC State Mutation

**CRITICAL**: All results are computed locally. AppState is ONLY mutated at the very end in Finalizing phase. This ensures no partial state on cancellation.

```rust
impl AppState {
    pub fn do_load_project_full(&self, path: PathBuf) -> Result<LoadProjectFullResult, LoadError> {
        let handle = self.get_app_handle()
            .ok_or_else(|| LoadError::internal("AppHandle not initialized"))?;
        
        // Get cancel token (cancels any previous load)
        let cancel_token = self.start_new_load();
        let mut progress = ProgressEmitter::new_for_open(handle, cancel_token);

        // Check already open BEFORE any work
        if self.is_same_project_open(&path) {
            self.clear_active_load();
            return Err(LoadError::already_open());
        }

        // === PHASE 0: Opening database (5%) ===
        progress.start_phase(LoadingPhase::OpeningDatabase);
        if !path.exists() {
            self.clear_active_load();
            return Err(LoadError::not_found(&path.display().to_string()));
        }
        // Local variable only - NO state mutation
        let db = Database::open_existing(&path)
            .map_err(|e| { self.clear_active_load(); LoadError::new("DATABASE", e.to_string()) })?;
        if progress.is_cancelled() { self.clear_active_load(); return Err(LoadError::cancelled()); }
        progress.complete_phase(LoadingPhase::OpeningDatabase);

        // === PHASE 1: Loading project data (20%) ===
        progress.start_phase(LoadingPhase::LoadingProjectData);
        let project = db.load_project()
            .map_err(|e| { self.clear_active_load(); LoadError::new("DATABASE", e.to_string()) })?;
        progress.update_phase(LoadingPhase::LoadingProjectData, 1, 1);
        if progress.is_cancelled() { self.clear_active_load(); return Err(LoadError::cancelled()); }
        progress.complete_phase(LoadingPhase::LoadingProjectData);

        // === PHASE 2: Loading sections (15%) ===
        progress.start_phase(LoadingPhase::LoadingSections);
        let sections = project.sections();
        let total_sections = sections.len() as u32;
        for (i, _section) in sections.iter().enumerate() {
            if progress.is_cancelled() { self.clear_active_load(); return Err(LoadError::cancelled()); }
            progress.update_phase(LoadingPhase::LoadingSections, i as u32 + 1, total_sections);
        }
        progress.complete_phase(LoadingPhase::LoadingSections);

        // === PHASE 3: Computing stations (20%) ===
        // Use EXISTING compute function with progress callback
        progress.start_phase(LoadingPhase::ComputingStations);
        let stations = compute_station_coordinates_with_progress(&project, |current, total| {
            if !progress.is_cancelled() {
                progress.update_phase(LoadingPhase::ComputingStations, current, total);
            }
        });
        if progress.is_cancelled() { self.clear_active_load(); return Err(LoadError::cancelled()); }
        progress.complete_phase(LoadingPhase::ComputingStations);

        // === PHASE 4: Computing statistics (15%) - COARSE-GRAINED ===
        progress.start_phase(LoadingPhase::ComputingStatistics);
        let stats = self.compute_stats_with_progress(&project, &mut progress)?;
        progress.complete_phase(LoadingPhase::ComputingStatistics);

        // === PHASE 5: Validating data (10%) ===
        progress.start_phase(LoadingPhase::ValidatingData);
        let warnings = self.validate_project(&project);
        if progress.is_cancelled() { self.clear_active_load(); return Err(LoadError::cancelled()); }
        progress.complete_phase(LoadingPhase::ValidatingData);

        // === PHASE 6: Loading settings (5%) ===
        progress.start_phase(LoadingPhase::LoadingSettings);
        let stored_settings = db.load_display_settings().unwrap_or_default();
        let display_settings = DisplaySettings::from_stored(&stored_settings);
        progress.complete_phase(LoadingPhase::LoadingSettings);

        // === PHASE 7: Finalizing (10%) - ONLY STATE MUTATION PHASE ===
        // Final cancel check BEFORE any state mutation
        if progress.is_cancelled() { self.clear_active_load(); return Err(LoadError::cancelled()); }
        
        progress.start_phase(LoadingPhase::Finalizing);
        
        // ATOMIC: All state updates happen here, after all computation is done
        *self.project.lock().unwrap() = Some(project.clone());
        self.add_to_recent(&path)
            .map_err(|e| { self.clear_active_load(); LoadError::new("RECENT", e) })?;
        
        progress.complete_phase(LoadingPhase::Finalizing);
        self.clear_active_load();

        Ok(LoadProjectFullResult {
            loading_id: progress.loading_id(),
            project,
            stations,
            stats,
            warnings,
            display_settings,
        })
    }
}
```

```rust
impl AppState {
    pub fn do_load_project_full(&self, path: PathBuf) -> Result<LoadProjectFullResult, LoadError> {
        let handle = self.get_app_handle()
            .ok_or_else(|| LoadError::internal("AppHandle not initialized"))?;

        // Get cancel token (cancels any previous load)
        let cancel_token = self.start_new_load();
        let mut progress = ProgressEmitter::new_for_open(handle, cancel_token);

        // Check already open
        if let Some(current) = self.project.lock().unwrap().as_ref() {
            if current.file_path == Some(path.clone()) {
                return Err(LoadError::already_open());
            }
        }

        // Phase 0: Opening database (5%)
        progress.start_phase(LoadingPhase::OpeningDatabase);
        if !path.exists() {
            return Err(LoadError::not_found(&path.display().to_string()));
        }
        let db = Database::open_existing(&path)
            .map_err(|e| LoadError::new("DATABASE", e.to_string()))?;
        if progress.is_cancelled() { return Err(LoadError::cancelled()); }
        progress.complete_phase(LoadingPhase::OpeningDatabase);

        // Phase 1: Loading project data (20%)
        progress.start_phase(LoadingPhase::LoadingProjectData);
        let project = db.load_project()
            .map_err(|e| LoadError::new("DATABASE", e.to_string()))?;
        progress.update_phase(LoadingPhase::LoadingProjectData, 1, 1);
        if progress.is_cancelled() { return Err(LoadError::cancelled()); }
        progress.complete_phase(LoadingPhase::LoadingProjectData);

        // Phase 2: Loading sections (15%)
        progress.start_phase(LoadingPhase::LoadingSections);
        let sections = project.sections();
        let total_sections = sections.len() as u32;
        for (i, _section) in sections.iter().enumerate() {
            if progress.is_cancelled() { return Err(LoadError::cancelled()); }
            progress.update_phase(LoadingPhase::LoadingSections, i as u32 + 1, total_sections);
        }
        progress.complete_phase(LoadingPhase::LoadingSections);

        // Phase 3: Computing stations (20%)
        // Total = sum(shots + 1) per section, NOT shots * 2
        progress.start_phase(LoadingPhase::ComputingStations);
        let total_stations: u32 = sections.iter()
            .map(|s| s.shots.len() as u32 + 1)
            .sum();
        let stations = self.compute_stations_with_progress(&project, &mut progress, total_stations)?;
        progress.complete_phase(LoadingPhase::ComputingStations);

        // Phase 4: Computing statistics (15%)
        progress.start_phase(LoadingPhase::ComputingStatistics);
        let stats = self.compute_stats_with_progress(&project, &mut progress)?;
        progress.complete_phase(LoadingPhase::ComputingStatistics);

        // Phase 5: Validating data (10%)
        progress.start_phase(LoadingPhase::ValidatingData);
        let warnings = self.validate_project(&project);
        if progress.is_cancelled() { return Err(LoadError::cancelled()); }
        progress.complete_phase(LoadingPhase::ValidatingData);

        // Phase 6: Loading settings (5%)
        progress.start_phase(LoadingPhase::LoadingSettings);
        let display_settings = db.load_display_settings()
            .map(|s| serde_json::to_value(s).unwrap_or_default())
            .unwrap_or_default();
        progress.complete_phase(LoadingPhase::LoadingSettings);

        // Phase 7: Finalizing (10%)
        progress.start_phase(LoadingPhase::Finalizing);
        *self.project.lock().unwrap() = Some(project.clone());
        self.add_to_recent(&path)
            .map_err(|e| LoadError::new("RECENT", e))?;
        progress.complete_phase(LoadingPhase::Finalizing);
        
        // Clear active loading
        *self.active_loading.lock().unwrap() = None;

        Ok(LoadProjectFullResult {
            loading_id: progress.loading_id(),
            project,
            stations,
            stats,
            warnings,
            display_settings,
        })
    }

    fn compute_stations_with_progress(
        &self,
        project: &Project,
        progress: &mut ProgressEmitter,
        total_stations: u32,
    ) -> Result<Vec<Station>, LoadError> {
        let mut stations = Vec::new();
        let mut station_count = 0u32;
        
        for section in project.sections() {
            for station in compute_section_stations(section) {
                if progress.is_cancelled() { return Err(LoadError::cancelled()); }
                station_count += 1;
                progress.update_phase(LoadingPhase::ComputingStations, station_count, total_stations);
                stations.push(station);
            }
        }
        
        Ok(stations)
    }
    
    fn compute_stats_with_progress(
        &self,
        project: &Project,
        progress: &mut ProgressEmitter,
    ) -> Result<ProjectStats, LoadError> {
        let sections = project.sections();
        let total = sections.len() as u32;
        
        for (i, _section) in sections.iter().enumerate() {
            if progress.is_cancelled() { return Err(LoadError::cancelled()); }
            progress.update_phase(LoadingPhase::ComputingStatistics, i as u32 + 1, total);
        }
        
        Ok(project.compute_stats())
    }
}
```
