## Modify: src/components/GlobalView/GlobalView.tsx

```tsx
import { LoadingProgress } from '../LoadingProgress';
import { useLoadingProgress } from '../../hooks/useLoadingProgress';

export function GlobalView() {
  const { loading, loadProject, resetLoading, cancelLoading } = useProjectStore();
  
  useLoadingProgress();

  const handleOpenProject = async (path: string) => {
    await loadProject(path);  // Use store method, not direct invoke
  };

  const handleCancel = async () => {
    await cancelLoading();
  };

  const handleRetry = async () => {
    if (loading.projectPath) {
      await loadProject(loading.projectPath);
    }
  };

  return (
    <div className="...">
      {loading.isLoading && (
        <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
          <LoadingProgress
            title={loading.error 
              ? 'Failed to Load Project' 
              : `Loading ${loading.projectName ?? 'project'}...`}
            phase={loading.phase}
            phaseIndex={loading.phaseIndex}
            phaseCount={loading.phaseCount}
            overallProgress={loading.overallProgress}
            onCancel={loading.error ? () => resetLoading() : handleCancel}
            onRetry={loading.error ? handleRetry : undefined}
            error={loading.error}
          />
        </div>
      )}

      <div className={loading.isLoading ? 'pointer-events-none opacity-50' : ''}>
        {/* existing content */}
      </div>
    </div>
  );
}
```

```tsx
import { LoadingProgress } from '../LoadingProgress';
import { useLoadingProgress } from '../../hooks/useLoadingProgress';
import { invoke } from '@tauri-apps/api/core';

export function GlobalView() {
  const { loading, resetLoading } = useProjectStore();
  
  // Set up Tauri event listener with timeout
  useLoadingProgress();

  const handleCancel = async () => {
    await invoke('cancel_loading');
    resetLoading();
  };

  const handleRetry = async () => {
    if (loading.projectPath) {
      resetLoading();
      await openProject(loading.projectPath);
    }
  };

  const handleDismiss = () => {
    resetLoading();
  };

  // Disable project cards while loading
  const isDisabled = loading.isLoading;

  return (
    <div className="...">
      {/* Modal loading overlay */}
      {loading.isLoading && (
        <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
          <LoadingProgress
            title={loading.error 
              ? 'Failed to Load Project' 
              : `Loading ${loading.projectName ?? 'project'}...`}
            phase={loading.phase ?? 'Initializing...'}
            phaseProgress={loading.phaseProgress}
            phaseCurrent={loading.phaseCurrent}
            phaseTotal={loading.phaseTotal}
            overallProgress={loading.overallProgress}
            onCancel={loading.error ? handleDismiss : handleCancel}
            onRetry={loading.error ? handleRetry : undefined}
            error={loading.error}
          />
        </div>
      )}

      {/* Project list/cards - disabled during loading */}
      <div className={isDisabled ? 'pointer-events-none opacity-50' : ''}>
        {/* ... existing GlobalView content */}
      </div>
    </div>
  );
}
```
