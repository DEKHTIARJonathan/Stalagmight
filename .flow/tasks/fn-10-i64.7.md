## Modify: src/components/GlobalView/GlobalView.tsx

Replace the simple "Loading..." text with LoadingProgress component:

```tsx
import { LoadingProgress } from '../LoadingProgress';
import { useLoadingProgress } from '../../hooks/useLoadingProgress';
import { invoke } from '@tauri-apps/api/core';

export function GlobalView() {
  const { loading, resetLoading } = useProjectStore();
  
  // Set up Tauri event listener
  useLoadingProgress();

  const handleCancel = async () => {
    await invoke('cancel_loading');
    resetLoading();
  };

  // ... existing code

  return (
    <div className="...">
      {loading.isLoading ? (
        <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
          <LoadingProgress
            title={`Loading ${loading.projectName ?? 'project'}...`}
            phase={loading.phase ?? 'Initializing...'}
            phaseProgress={loading.phaseProgress}
            phaseCurrent={loading.phaseCurrent}
            phaseTotal={loading.phaseTotal}
            overallProgress={loading.overallProgress}
            onCancel={handleCancel}
            error={loading.error}
          />
        </div>
      ) : (
        // ... existing GlobalView content
      )}
    </div>
  );
}
```

## Key Changes

3. Replace `{isLoading ? <div>Loading...</div> : ...}` with LoadingProgress
4. Add semi-transparent backdrop for modal behavior
5. Wire up cancel button to invoke cancel_loading command
6. Pass all loading state from store to component
