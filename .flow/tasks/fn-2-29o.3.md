export async function resetAppState(): Promise<void> {
  // Close dialog and project via store - closeProject is ASYNC
  await browser.execute(async () => {
    const e2e = (window as any).__E2E_TEST__;
    if (e2e) {
      e2e.uiStore.getState().closeDialog();
      await e2e.projectStore.getState().closeProject();
      e2e.uiStore.getState().setView('global');
    }
  });

  // Wait for global view to confirm reset
  await browser.waitUntil(
    async () => {
      const globalView = await $('[data-testid="global-view"]');
      return await globalView.isExisting();
    },
    { timeout: 5000, timeoutMsg: 'Failed to return to GlobalView after reset' }
  );
}

### 2. Rename and REWRITE import-dialog.spec.ts

Rename to `import-dmp-dialog.spec.ts` AND rewrite tests to assert DmpImportDialog UI:

```typescript
// import-dmp-dialog.spec.ts - Tests for DMP file import dialog (Ctrl+I)
describe('DMP Import Dialog', () => {
  beforeEach(async () => {
    await waitForAppReady();
    await resetAppState();
  });

  it('should open DMP import dialog with Ctrl+I', async () => {
    await createNewProject('DMP Import Test');
    await browser.keys(['Control', 'i']);

    const dialog = await $('[data-testid="import-dmp-dialog"]');
    await dialog.waitForDisplayed({ timeout: 5000 });

    // Verify DmpImportDialog-specific UI elements
    const fileInput = await dialog.$('input[type="text"][readonly]');
    await expect(fileInput).toBeExisting();

    const browseButton = await dialog.$('button*=Browse');
    await expect(browseButton).toBeExisting();

    // Close with Cancel
    const cancelBtn = await dialog.$('button*=Cancel');
    await cancelBtn.click();
    await dialog.waitForDisplayed({ reverse: true, timeout: 5000 });
  });
});
```

### 3. Create import-section-dialog.spec.ts

New spec for testing ImportDialog (section name input):

```typescript
// import-section-dialog.spec.ts
describe('Import Section Dialog', () => {
  beforeEach(async () => {
    await waitForAppReady();
    await resetAppState();
  });

  it('should display section name input field', async () => {
    await createNewProject('Section Name Test');

    // Wait for E2E store
    await browser.waitUntil(
      async () => await browser.execute(() => !!(window as any).__E2E_TEST__),
      { timeout: 5000 }
    );

    // Open ImportDialog (not DmpImportDialog)
    await browser.execute(() => {
      (window as any).__E2E_TEST__.uiStore.getState().openDialog('import');
    });

    // Wait for dialog with reasonable timeout
    const dialog = await $('[data-testid="import-section-dialog"]');
    await dialog.waitForDisplayed({ timeout: 5000 });

    // Validate input
    const sectionNameInput = await $('[data-testid="section-name-input"]');
    await expect(sectionNameInput).toBeExisting();

    const placeholder = await sectionNameInput.getAttribute('placeholder');
    expect(placeholder).toBe('e.g., Main Line');

    await sectionNameInput.setValue('Main Passage');
    const value = await sectionNameInput.getValue();
    expect(value).toBe('Main Passage');

    // Close with Cancel - use waitForDisplayed reverse instead of waitUntil
    const cancelBtn = await dialog.$('button*=Cancel');
    await cancelBtn.click();
    await dialog.waitForDisplayed({ reverse: true, timeout: 5000 });
  });
});
```
