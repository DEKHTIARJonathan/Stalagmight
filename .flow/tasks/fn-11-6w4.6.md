# fn-11-6w4.6 Add color mode and preset selectors to ExportDialog

## Description
Add color mode, preset, and legend selectors to ExportDialog.

**CRITICAL: Use useEffect + invoke, NOT react-query (not in project).**

## Fetch Preset Colors
```tsx
// DON'T use react-query - it's not in the project
// const { data } = useQuery(...) // WRONG

// DO use useEffect + invoke
const [presetColors, setPresetColors] = useState<{ shallow: string; deep: string } | null>(null);

useEffect(() => {
  if (imageOptions.color_mode === 'depth') {
    invoke<{ shallow: string; deep: string }>('get_depth_preset_colors', { 
      preset: imageOptions.depth_preset 
    })
      .then(setPresetColors)
      .catch(err => console.error('Failed to fetch preset colors:', err));
  } else {
    setPresetColors(null);
  }
}, [imageOptions.color_mode, imageOptions.depth_preset]);
```

### New State
```typescript
// Preset colors fetched from backend
const [presetColors, setPresetColors] = useState<{ shallow: string; deep: string } | null>(null);

// Fetch preset colors when depth mode and preset changes
useEffect(() => {
  if (imageOptions.color_mode === 'depth') {
    invoke<{ shallow: string; deep: string }>('get_depth_preset_colors', {

    }).then(setPresetColors);
```

### Color Mode Section (after existing options)
```tsx
{exportType === 'image' && (
  <>
    {/* ... existing format and options ... */}
    
    {/* Color Mode */}
    <div>
      <label className="block text-sm text-text-secondary mb-2">Color Mode</label>
      <div className="flex gap-4">
        <label className="flex items-center gap-2">
          <input
            type="radio"
            checked={imageOptions.color_mode === 'section'}
            onChange={() => setImageOptions({ ...imageOptions, color_mode: 'section' })}
            data-testid="color-mode-section"
          />
          <span>Section Colors</span>
        </label>
        <label className="flex items-center gap-2">
          <input
            type="radio"
            checked={imageOptions.color_mode === 'depth'}
            onChange={() => setImageOptions({ ...imageOptions, color_mode: 'depth' })}
            data-testid="color-mode-depth"
          />
          <span>Depth Gradient</span>
        </label>
      </div>
    </div>
    
    {/* Depth Options (only when depth mode) */}
    {imageOptions.color_mode === 'depth' && (
      <>
        <div>
          <label className="block text-sm text-text-secondary mb-2">Color Preset</label>
          <Select
            value={imageOptions.depth_preset}
            onChange={(v) => setImageOptions({ ...imageOptions, depth_preset: v as DepthPreset })}
            options={[
              { value: 'thermal', label: 'Thermal (Blue → Red)' },
              { value: 'ocean', label: 'Ocean (Cyan → Blue)' },
              { value: 'terrain', label: 'Terrain (Green → Brown)' },
              { value: 'grayscale', label: 'Grayscale' },
              { value: 'accessible', label: 'Accessible (Yellow → Purple)' },
            ]}
            data-testid="depth-preset-select"
          />
        </div>
        
        <label className="flex items-center gap-2">
          <input
            type="checkbox"
            checked={imageOptions.include_legend}
            onChange={(e) => setImageOptions({ ...imageOptions, include_legend: e.target.checked })}
            data-testid="option-include-legend"
          />
          <span>Include depth legend</span>
        </label>
      </>
    )}
    
    {/* Resolution - add 600 DPI */}
    {imageFormat === 'png' && (
      <div>
        <label className="block text-sm text-text-secondary mb-2">Resolution</label>
        <Select
          value={String(imageOptions.dpi)}
          onChange={(v) => setImageOptions({ ...imageOptions, dpi: parseInt(v) })}
          options={[
            { value: '72', label: 'Low (72 DPI)' },
            { value: '150', label: 'Medium (150 DPI)' },
            { value: '300', label: 'High (300 DPI)' },
            { value: '600', label: 'Print (600 DPI)' },  // NEW
          ]}
          data-testid="resolution-select"
        />
      </div>
    )}
  </>
)}
```
