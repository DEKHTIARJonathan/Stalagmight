**CRITICAL: interpolate_stops safety guards:**
```rust
fn interpolate_stops(t: f64, stops: &[ColorStop]) -> String {
    // Guard for empty or single-element
    if stops.is_empty() {
        return "#000000".to_string();  // Fallback black
    }
    if stops.len() == 1 {
        return stops[0].color.clone();
    }
    
    // Clamp t to valid range
    let t = t.clamp(0.0, 1.0);
    
    // Find bracketing stops (assumes sorted by position)
    for i in 0..stops.len() - 1 {
        if t >= stops[i].position && t <= stops[i + 1].position {
            // Handle equal positions (avoid divide by zero)
            if (stops[i + 1].position - stops[i].position).abs() < f64::EPSILON {
                return stops[i].color.clone();
            }
            let local_t = (t - stops[i].position) / (stops[i + 1].position - stops[i].position);
            return interpolate_rgb(&stops[i].color, &stops[i + 1].color, local_t);
        }
    }
    stops.last().unwrap().color.clone()
}
```

**Tauri command returns full stop list:**

fn format_depth_label(depth: f64, unit: &str, precision: u8) -> String {
    let value = if unit == "ft" {
        depth * 3.28084  // meters to feet conversion
    } else {
        depth
    };
    format!("{:.prec$}{}", value, unit, prec = precision as usize)
}

export interface ColorStop {
  position: number;
  color: string;
}

export function interpolateStops(t: number, stops: ColorStop[]): string {
  // Guard for empty or single-element (match Rust backend)
  if (stops.length === 0) return '#000000';
  if (stops.length === 1) return stops[0].color;
  
  const clampedT = Math.max(0, Math.min(1, t));
  
  for (let i = 0; i < stops.length - 1; i++) {
    if (clampedT >= stops[i].position && clampedT <= stops[i + 1].position) {
      const denominator = stops[i + 1].position - stops[i].position;
      if (Math.abs(denominator) < Number.EPSILON) {
        return stops[i].color;
      }
      const localT = (clampedT - stops[i].position) / denominator;
      return interpolateRgb(stops[i].color, stops[i + 1].color, localT);
    }
  }
  return stops[stops.length - 1].color;
}

## Edge Cases

| Case | Handling |
|------|----------|
| All stations at same depth | All get first stop color, legend shows "X.Xm (Flat)" |
| min === max | t=0, returns first stop color |
| z < min | Clamp t to 0 |
| z > max | Clamp t to 1 |
| NaN/infinite z | **Explicit is_finite() check**, returns first stop color |
| Invalid preset | **Error** - Rust enum fails deserialization |
| Export size > 16384px | **Error** with clear message |
| Export cancelled | **Not supported** - sync command |
| Unit = "ft" | **Convert value** (meters * 3.28084), not just suffix |
| Empty stops array | Return fallback "#000000" |
| Single stop | Return that stop's color |
| Equal positions | Return first stop's color (avoid divide by zero) |
| config is null | Use defaults: unit="m", precision=1 |
