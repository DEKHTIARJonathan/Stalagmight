### 4. Key Discovery: MapLibre Timing Quirk
When `dragPan` is disabled at map creation and later enabled, the drag handlers need "warmup":

```typescript
async function ctrlDragWithWarmup(page, startX, startY, endX, endY) {
  await page.keyboard.down('Control');
  await page.waitForTimeout(150);
  
  // Warmup: trigger panBy to initialize handlers
  await page.evaluate(() => {
    const map = window.__stalagmight_map__;
    if (map) {
      map.panBy([100, 0], { animate: false });
      map.panBy([-100, 0], { animate: false });
      map.panBy([50, 0], { animate: false });
      map.panBy([-50, 0], { animate: false });
    }
  });
  await page.waitForTimeout(150);
  
  // Actual drag
  await page.mouse.move(startX, startY);
  await page.mouse.down();
  await page.mouse.move(endX, endY, { steps: 10 });
  await page.mouse.up();
  await page.keyboard.up('Control');
}
```
